1//! Original first-recording work geometry and device-produced work. Units are
2//! logical operations, not elapsed time, instructions or FLOPs. Every recorded
3//! occurrence is retained; conditional producers supply actual units separately
4//! from their cold upper geometry.
5
6use std::collections::HashSet;
7
8#[repr(u64)]
9#[derive(Clone, Copy, Debug, PartialEq, Eq)]
10pub enum ModelWorkKind {
11    Pointwise = 1,
12    Reduction = 2,
13    Scan = 3,
14    Fill = 4,
15    Contraction = 5,
16    Softmax = 6,
17    LogSoftmax = 7,
18    LogSumExp = 8,
19    Scatter = 9,
20    Gather = 10,
21    CopyBytes = 11,
22    Cast = 12,
23    SortKeys = 13,
24    BucketQueries = 14,
25    SavedCopyBytes = 15,
26}
27
28impl ModelWorkKind {
29    pub fn from_code(code: u64) -> Option<Self> {
30        Some(match code {
31            1 => Self::Pointwise,
32            2 => Self::Reduction,
33            3 => Self::Scan,
34            4 => Self::Fill,
35            5 => Self::Contraction,
36            6 => Self::Softmax,
37            7 => Self::LogSoftmax,
38            8 => Self::LogSumExp,
39            9 => Self::Scatter,
40            10 => Self::Gather,
41            11 => Self::CopyBytes,
42            12 => Self::Cast,
43            13 => Self::SortKeys,
44            14 => Self::BucketQueries,
45            15 => Self::SavedCopyBytes,
46            _ => return None,
47        })
48    }
49}

51/// Immutable kernel input. Unused dimensions are zero, while a scalar has rank
52/// zero and one unit. Saved-copy coordinates name the original content witness
53/// and tensor occurrence; ordinary operations have no saved witness.
54#[repr(C)]
55#[derive(Clone, Copy, Debug, PartialEq, Eq)]
56pub(crate) struct ModelWorkEvent {
57    pub kind: u64,
58    pub witness: u64,
59    pub tensor: u64,
60    pub rank: u64,
61    pub dimensions: [u64; 8],
62    /// Zero for exact geometry, otherwise the original retained device slot
63    /// containing [actual units, sticky overflow, producer writes].
64    pub actual: u64,
65}
66
67// SAFETY: the CUDA ABI is thirteen padding-free u64 words, with no host pointers.
68unsafe impl crate::DeviceRepr for ModelWorkEvent {}
69const _: () = assert!(std::mem::size_of::<ModelWorkEvent>() == 104);
70
71impl ModelWorkEvent {
72    pub(crate) fn operation(kind: ModelWorkKind, dimensions: &[u64]) -> Result<Self, &'static str> {
73        if kind == ModelWorkKind::SavedCopyBytes {
74            return Err("saved-copy work requires its original tensor witness");
75        }
76        Self::new(kind, u64::MAX, u64::MAX, dimensions)
77    }
78
79    pub(crate) fn device_operation(
80        kind: ModelWorkKind,
81        upper_dimensions: &[u64],
82        actual: u64,
83    ) -> Result<Self, &'static str> {
84        if actual == 0 || !actual.is_multiple_of(8) || actual > u64::MAX - 24 {
85            return Err("model work requires its complete aligned native device slot");
86        }
87        let mut event = Self::operation(kind, upper_dimensions)?;
88        event.actual = actual;
89        Ok(event)
90    }
91
92    pub(crate) fn saved(
93        witness: u64,
94        tensor: u64,
95        dimensions: &[u64],
96        element_bytes: u64,
97    ) -> Result<Self, &'static str> {
98        if witness == u64::MAX
99            || tensor == u64::MAX
100            || dimensions.len() >= 8
101            || !matches!(element_bytes, 1 | 2 | 4 | 8)
102        {
103            return Err("saved-copy work has invalid original tensor geometry");
104        }
105        let mut shape = [0; 8];
106        shape[..dimensions.len()].copy_from_slice(dimensions);
107        shape[dimensions.len()] = element_bytes;
108        Self::new(
109            ModelWorkKind::SavedCopyBytes,
110            witness,
111            tensor,
112            &shape[..dimensions.len() + 1],
113        )
114    }
115
116    fn new(
117        kind: ModelWorkKind,
118        witness: u64,
119        tensor: u64,
120        dimensions: &[u64],
121    ) -> Result<Self, &'static str> {
122        if dimensions.len() > 8 {
123            return Err("model work geometry exceeds eight dimensions");
124        }
125        let mut event = Self {
126            kind: kind as u64,
127            witness,
128            tensor,
129            rank: dimensions.len() as u64,
130            dimensions: [0; 8],
131            actual: 0,
132        };
133        event.dimensions[..dimensions.len()].copy_from_slice(dimensions);
134        event.units()?;
135        Ok(event)
136    }
137
138    fn units(&self) -> Result<u64, &'static str> {
139        let kind = ModelWorkKind::from_code(self.kind).ok_or("model work unit kind is unknown")?;
140        if self.rank > 8
141            || self.dimensions[self.rank as usize..]
142                .iter()
143                .any(|&extent| extent != 0)
144            || (kind == ModelWorkKind::SavedCopyBytes
145                && (self.witness == u64::MAX
146                    || self.tensor == u64::MAX
147                    || self.actual != 0
148                    || self.rank == 0
149                    || !matches!(self.dimensions[self.rank as usize - 1], 1 | 2 | 4 | 8)))
150            || (kind != ModelWorkKind::SavedCopyBytes
151                && (self.witness != u64::MAX || self.tensor != u64::MAX))
152        {
153            return Err("model work event has noncanonical geometry or occurrence coordinates");
154        }
155        let dimensions = &self.dimensions[..self.rank as usize];
156        if dimensions.contains(&0) {
157            return Ok(0);
158        }
159        dimensions
160            .iter()
161            .try_fold(1u64, |units, &extent| units.checked_mul(extent))
162            .ok_or("model work geometry exceeds the exact u64 counter")
163    }
164}

166/// Capacity is reserved cold. Neither host vector growth nor device allocation
167/// is allowed when original producer callbacks append their first-record events.
168pub(crate) struct ModelWorkRecording {
169    capacity: usize,
170    events: Vec<ModelWorkEvent>,
171    saved: HashSet<(u64, u64)>,
172    bound: u64,
173    minimum: u64,
174    frozen: bool,
175}
176
177impl ModelWorkRecording {
178    pub(crate) fn new(capacity: usize) -> Result<Self, &'static str> {
179        if capacity == 0 {
180            return Err("model work requires a positive cold event capacity");
181        }
182        let mut events = Vec::new();
183        events
184            .try_reserve_exact(capacity)
185            .map_err(|_| "model work event reservation failed")?;
186        let mut saved = HashSet::new();
187        saved
188            .try_reserve(capacity)
189            .map_err(|_| "saved occurrence reservation failed")?;
190        Ok(Self {
191            capacity,
192            events,
193            saved,
194            bound: 0,
195            minimum: 0,
196            frozen: false,
197        })
198    }
199
200    pub(crate) fn push(&mut self, event: ModelWorkEvent) -> Result<(), &'static str> {
201        if self.frozen {
202            return Err("original model work is already frozen");
203        }
204        if self.events.len() == self.capacity {
205            return Err("model work exceeds its cold event capacity");
206        }
207        let bound = self
208            .bound
209            .checked_add(event.units()?)
210            .ok_or("model work sum exceeds the exact u64 counter")?;
211        let saved = event.kind == ModelWorkKind::SavedCopyBytes as u64;
212        if saved && self.saved.contains(&(event.witness, event.tensor)) {
213            return Err("saved-tensor occurrence was already charged");
214        }
215        self.events.push(event);
216        if saved {
217            self.saved.insert((event.witness, event.tensor));
218        }
219        if event.actual == 0 {
220            // This exact contribution is a subset of the checked total bound.
221            self.minimum += bound - self.bound;
222        }
223        self.bound = bound;
224        Ok(())
225    }
226
227    #[cfg(any(test, feature = "semantic-policy"))]
228    pub(crate) fn freeze(&mut self) -> Result<u64, &'static str> {
229        if self.frozen || self.events.is_empty() || self.bound == 0 {
230            return Err("model work requires one nonempty original recording and a single freeze");
231        }
232        self.frozen = true;
233        Ok(self.bound)
234    }
235
236    pub(crate) fn events(&self) -> &[ModelWorkEvent] {
237        &self.events
238    }
239    pub(crate) fn frozen_bound(&self) -> Option<u64> {
240        self.frozen.then_some(self.bound)
241    }
242}

244/// Checked device accumulator. The model and native contributions are kept
245/// separate; neither the old edit facts nor an estimated capacity is a charge.
246#[repr(C)]
247#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
248pub(crate) struct ExecutionWork {
249    pub raw: u64,
250    pub model_once: u64,
251    pub native_attempt: u64,
252    pub model_events: u64,
253    pub model_bound: u64,
254    pub overflow: u64,
255    /// Fixed native event order follows the producer's unit-tariff registry.
256    pub native_events: [u64; 9],
257    /// Attribution of existing native units; never added a second time to raw.
258    pub candidates: [u64; 3],
259    /// Zero is shared work; one through three select an original candidate.
260    pub active_candidate: u64,
261}
262
263impl ExecutionWork {
264    #[cfg(any(test, feature = "semantic-policy"))]
265    pub(crate) fn validate_model(
266        &self,
267        recording: &ModelWorkRecording,
268        refused_overflow: bool,
269    ) -> bool {
270        recording.frozen_bound() == Some(self.model_bound)
271            && self.model_once.checked_add(self.native_attempt) == Some(self.raw)
272            && self
273                .native_events
274                .iter()
275                .try_fold(0u64, |sum, value| sum.checked_add(*value))
276                == Some(self.native_attempt)
277            && self
278                .candidates
279                .iter()
280                .try_fold(0u64, |sum, value| sum.checked_add(*value))
281                .is_some_and(|sum| sum <= self.native_attempt)
282            && self.active_candidate == 0
283            && if refused_overflow {
284                self.overflow == 1 && self.model_events <= recording.events().len() as u64
285            } else {
286                self.overflow == 0
287                    && self.model_events == recording.events().len() as u64
288                    && (recording.minimum..=self.model_bound).contains(&self.model_once)
289            }
290    }
291}

299    #[test]
300    fn event_geometry_counts_declared_units_without_dense_mask_discount() {
301        let event = ModelWorkEvent::operation(ModelWorkKind::Contraction, &[2, 3, 5]).unwrap();
302        assert_eq!(event.units().unwrap(), 30);
303        assert_eq!(
304            ModelWorkEvent::operation(ModelWorkKind::Pointwise, &[])
305                .unwrap()
306                .units()
307                .unwrap(),
308            1
309        );
310        assert_eq!(
311            ModelWorkEvent::operation(ModelWorkKind::Gather, &[4, 0, 7])
312                .unwrap()
313                .units()
314                .unwrap(),
315            0
316        );
317        assert!(ModelWorkEvent::operation(ModelWorkKind::Contraction, &[u64::MAX, 2]).is_err());
318        assert!(ModelWorkEvent::operation(ModelWorkKind::Pointwise, &[1; 9]).is_err());
319    }

321    #[test]
322    fn saved_occurrences_keep_original_witness_coordinates_and_byte_extents() {
323        let mut recording = ModelWorkRecording::new(3).unwrap();
324        recording
325            .push(ModelWorkEvent::saved(7, 0, &[2, 3], 4).unwrap())
326            .unwrap();
327        recording
328            .push(ModelWorkEvent::saved(8, 0, &[2, 3], 4).unwrap())
329            .unwrap();
330        assert!(recording
331            .push(ModelWorkEvent::saved(7, 0, &[2, 3], 4).unwrap())
332            .is_err());
333        assert_eq!(recording.freeze().unwrap(), 48);
334        assert_eq!(recording.events()[0].witness, 7);
335        assert_eq!(recording.events()[1].witness, 8);
336    }

338    #[test]

340        let mut recording = ModelWorkRecording::new(1).unwrap();

342        recording.push(event).unwrap();

344        assert_eq!(recording.freeze().unwrap(), 12);
345        assert!(recording.freeze().is_err());
346        assert!(recording.push(event).is_err());
347        let mut overflow = ModelWorkRecording::new(2).unwrap();
348        overflow
349            .push(ModelWorkEvent::operation(ModelWorkKind::CopyBytes, &[u64::MAX]).unwrap())
350            .unwrap();
351        assert!(overflow
352            .push(ModelWorkEvent::operation(ModelWorkKind::Fill, &[1]).unwrap())
353            .is_err());
354        assert!(ModelWorkRecording::new(0).is_err());

358    #[test]
359    fn completion_consumes_the_original_frozen_geometry_and_checked_common_sum() {
360        let mut recording = ModelWorkRecording::new(1).unwrap();
361        recording
362            .push(ModelWorkEvent::operation(ModelWorkKind::Contraction, &[3, 7]).unwrap())
363            .unwrap();
364        let mut actual = ExecutionWork {
365            model_once: 21,
366            native_attempt: 5,
367            raw: 26,
368            model_events: 1,
369            model_bound: 21,
370            native_events: [5, 0, 0, 0, 0, 0, 0, 0, 0],
371            ..ExecutionWork::default()
372        };
373        assert!(!actual.validate_model(&recording, false));
374        recording.freeze().unwrap();
375        assert!(actual.validate_model(&recording, false));
376        actual.raw = 21;
377        assert!(!actual.validate_model(&recording, false));
378        actual.raw = 26;
379        actual.model_events = 0;
380        assert!(!actual.validate_model(&recording, false));
381        actual.overflow = 1;
382        assert!(actual.validate_model(&recording, true));
383        actual.model_bound = 22;
384        assert!(!actual.validate_model(&recording, true));
385    }

389        let mut recording = ModelWorkRecording::new(1).unwrap();
