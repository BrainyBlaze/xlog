9/// Free-Join atom processing-order decision (decline-or-reorder).
10///
11/// The Free Join engine materializes a left-deep prefix; because each probe's
12/// keys must be a leading *column* prefix of its atom, it cannot reorder a
13/// chain to start from a selective tail the way the binary fallback can. On an
14/// adversarial chain this forces a large intermediate even when the result is
15/// tiny (a measured worst case is roughly a 3x peak-memory loss vs binary).
16/// This decision lets the dispatcher rescue or decline that case.
17pub(super) enum FjOrderDecision {
18    /// Keep the default traversal order. Returned whenever the current order
19    /// is already estimated within tolerance of the binary plan (so every
20    /// winning fixture is untouched) or the planner is inactive — i.e. the
21    /// dispatch behaves exactly as it did before the order planner existed
22    /// (fail-OPEN).
23    KeepDefault,
24    /// Dispatch Free Join with this atom processing order (original input
25    /// indices). Prefix-key-joinable by construction.
26    Reorder(Vec<usize>),
27    /// No prefix-key-joinable order is estimated within tolerance of the
28    /// unconstrained binary plan's peak — decline FJ so the binary fallback
29    /// runs (fail-CLOSED).
30    Decline,
31}
32
33/// Free Join is reordered/declined only when the current order's estimated
34/// peak exceeds the binary plan's by more than this ratio (6/5 = 1.2×).
35/// Integer math (`5·a ≤ 6·b` ⇔ `a ≤ 1.2·b`) avoids float comparison. Mirrors
36/// the adversarial-chain peak-memory acceptance test in the integration suite.
37const FJ_PEAK_TOLERANCE_NUM: u128 = 6;
38const FJ_PEAK_TOLERANCE_DEN: u128 = 5;
39
40/// `true` iff `peak ≤ 1.2 × baseline` (within tolerance — FJ is acceptable).
41fn within_peak_tolerance(peak: u64, baseline: u64) -> bool {
42    (peak as u128) * FJ_PEAK_TOLERANCE_DEN <= (baseline as u128) * FJ_PEAK_TOLERANCE_NUM
43}
44
45/// Default selectivity StatsManager applies when no cached selectivity /
46/// column-distinct stats exist (10%). Reused for the statless local estimate
47/// so the planner works for single-rule base-relation joins (where scans
48/// never populate `StatsManager`), consistent with the authoritative path.
49const DEFAULT_JOIN_SELECTIVITY_DEN: u128 = 10;

51pub(super) trait WcojCostModel: Send + Sync {
52    fn should_dispatch_triangle(&self, ctx: &WcojDispatchCtx) -> bool;
53    fn should_dispatch_4cycle(&self, ctx: &WcojDispatchCtx) -> bool;
54
55    /// Fail-OPEN loss-region veto for the general factorized routes
56    /// (aggregate-fused WCOJ over a triangle; Free Join). Returns
57    /// `true` ONLY when the model has full cardinality stats for every
58    /// slot relation AND the largest one is below the WCOJ-worthwhile
59    /// threshold — i.e. the join is provably small, no intermediate can
60    /// blow up, and the binary fallback wins (the measured 1.7–2.0×
61    /// cost-of-generality region for Free Join, and the small-triangle
62    /// region the base triangle cost model already declines).
63    ///
64    /// Deliberately conservative: stats absent for ANY slot, or ANY slot
65    /// large → `false` (no veto). This NEVER vetoes a case with a large
66    /// input (where factorized can win on a large avoided intermediate)
67    /// and NEVER vetoes when stats are unavailable (e.g. recursive deltas
68    /// on early iterations) — so every measured factorized gate win is
69    /// preserved exactly; the veto only removes provably-small losses.
70    fn factorized_loss_veto(&self, ctx: &WcojDispatchCtx) -> bool {
71        let _ = ctx;
72        false
73    }
74
75    /// Free Join order planner (decline-or-reorder). Acts only as a safety
76    /// net: when the current traversal order is already estimated within
77    /// tolerance of the binary plan it returns [`FjOrderDecision::KeepDefault`]
78    /// (every winning fixture untouched); only when the current order is
79    /// estimated to LOSE does it search for a better prefix-key-joinable order,
80    /// reordering to it ([`FjOrderDecision::Reorder`]) or declining to the
81    /// binary fallback ([`FjOrderDecision::Decline`]) if none is competitive.
82    ///
83    /// `atom_vars[i]` = dense join-variable ids per column of input `i` (column
84    /// order preserved — the prefix-key constraint keys on it); `cards[i]` =
85    /// that input's ground-truth row count (from the buffer the dispatcher is
86    /// about to join, NOT `StatsManager` — so it is always available even for
87    /// single-rule base-relation joins); `ctx.slot_rels[i]` = its relation.
88    /// All three index spaces coincide. Per-pair join estimates use
89    /// `StatsManager::estimate_join_cardinality` when stats are populated.
90    ///
91    /// Default: [`FjOrderDecision::KeepDefault`] — only `CardinalityAware`
92    /// plans (so the `SkewClassifier` opt-out also disables the reorder).
93    fn plan_free_join_order(
94        &self,
95        ctx: &WcojDispatchCtx,
96        atom_vars: &[Vec<usize>],
97        cards: &[u64],
98    ) -> FjOrderDecision {
99        let _ = (ctx, atom_vars, cards);
100        FjOrderDecision::KeepDefault
101    }
102}

121    min_binary_intermediate: u64,
122    large_binary_intermediate: u64,
123}

125impl Default for CardinalityAwareCostModel {
126    fn default() -> Self {
127        Self {
128            min_binary_intermediate: MIN_CARDINALITY_BINARY_INTERMEDIATE,
129            large_binary_intermediate: LARGE_CARDINALITY_BINARY_INTERMEDIATE,
130        }
131    }
132}

134impl CardinalityAwareCostModel {
135    fn populated_cards(&self, ctx: &WcojDispatchCtx) -> Option<Vec<u64>> {
136        ctx.slot_rels
137            .iter()
138            .map(|r| {
139                ctx.stats
140                    .get_relation_stats(*r)
141                    .map(|s| s.cardinality)
142                    .filter(|c| *c > 0)
143            })
144            .collect()
145    }
146
147    fn decide_from_cardinality(&self, binary_est: u64) -> bool {
148        binary_est >= self.large_binary_intermediate || binary_est >= self.min_binary_intermediate
149    }
150}

152impl WcojCostModel for CardinalityAwareCostModel {
153    fn should_dispatch_triangle(&self, ctx: &WcojDispatchCtx) -> bool {
154        debug_assert_eq!(
155            ctx.slot_rels.len(),
156            3,
157            "triangle ctx must carry exactly 3 slot relations"
158        );
159        if self.populated_cards(ctx).is_none() {
160            return false;
161        }
162        let binary_est =
163            ctx.stats
164                .estimate_join_cardinality(ctx.slot_rels[0], ctx.slot_rels[1], &[1], &[0]);
165        self.decide_from_cardinality(binary_est)
166    }
167
168    fn should_dispatch_4cycle(&self, ctx: &WcojDispatchCtx) -> bool {
169        debug_assert_eq!(
170            ctx.slot_rels.len(),
171            4,
172            "4-cycle ctx must carry exactly 4 slot relations"
173        );
174        if self.populated_cards(ctx).is_none() {
175            return false;
176        }
177        let binary_est =
178            ctx.stats
179                .estimate_join_cardinality(ctx.slot_rels[0], ctx.slot_rels[1], &[1], &[0]);
180        self.decide_from_cardinality(binary_est)
181    }
182
183    fn factorized_loss_veto(&self, ctx: &WcojDispatchCtx) -> bool {
184        // Fail-open: need every slot's cardinality to make any claim.
185        let cards = match self.populated_cards(ctx) {
186            Some(c) => c,
187            None => return false,
188        };
189        // Veto only when the LARGEST input is below the WCOJ-worthwhile
190        // threshold: then every join intermediate is bounded small, the
191        // binary plan is cheap, and the factorized route's overhead is
192        // not justified. Any large input → no veto (factorized may win
193        // on the avoided large intermediate).
194        cards.iter().copied().max().unwrap_or(0) < self.min_binary_intermediate
195    }
196
197    fn plan_free_join_order(
198        &self,
199        ctx: &WcojDispatchCtx,
200        atom_vars: &[Vec<usize>],
201        cards: &[u64],
202    ) -> FjOrderDecision {
203        let n = atom_vars.len();
204        // Defensive: index spaces must coincide; <3 atoms never reach here.
205        if n < 3 || cards.len() != n || ctx.slot_rels.len() != n {
206            return FjOrderDecision::KeepDefault;
207        }
208        // Binary fallback's freedom: best estimated peak over ANY connected
209        // order (no prefix-key constraint). The yardstick both the current
210        // and any candidate FJ order are judged against.
211        let binary_peak = match best_greedy_peak(ctx, atom_vars, cards, false) {
212            Some((_, p)) => p,
213            // No connected order at all (degenerate) → don't intervene.
214            None => return FjOrderDecision::KeepDefault,
215        };
216        // Safety-net gate: if the CURRENT traversal order is already within
217        // tolerance, keep it — guarantees every winning fixture (whose
218        // current order is good) is byte-for-byte untouched.
219        let traversal: Vec<usize> = (0..n).collect();
220        match eval_order_peak(ctx, atom_vars, cards, &traversal, true) {
221            Some(default_peak) => {
222                // Absolute floor: a small intermediate is cheap regardless of
223                // order — never intervene (the loss veto's small-join
224                // territory; FJ fires exactly as before). Only LARGE
225                // intermediates can carry the order-loss the planner removes.
226                // Also avoids spurious declines from ratio noise on tiny cards.
227                if default_peak < self.min_binary_intermediate
228                    || within_peak_tolerance(default_peak, binary_peak)
229                {
230                    return FjOrderDecision::KeepDefault;
231                }
232            }
233            None => {
234                // Traversal order is not prefix-key-joinable (FJ already
235                // declines to binary). Only attempt a rescue when the join is
236                // large enough to matter; otherwise keep the original behavior.
237                if binary_peak < self.min_binary_intermediate {
238                    return FjOrderDecision::KeepDefault;
239                }
240            }
241        }
242        // Current order loses (or is not prefix-key-valid) on a large join.
243        // Try the best prefix-key-joinable order; reorder to it if competitive,
244        // else decline to the binary fallback.
245        match best_greedy_peak(ctx, atom_vars, cards, true) {
246            Some((order, fj_peak)) if within_peak_tolerance(fj_peak, binary_peak) => {
247                FjOrderDecision::Reorder(order)
248            }
249            _ => FjOrderDecision::Decline,
250        }
251    }
252}

261// ---------------------------------------------------------------------------
262// Free Join order-planning helpers (decline-or-reorder).
263//
264// A left-deep prefix model: starting from a leader relation, atoms are added
265// one at a time; the running intermediate size is multiplied by the per-tuple
266// fan-out of each added atom (independence assumption), and the planner tracks
267// the peak running size. `constrained = true` enforces FJ's rule that an
268// atom's already-bound vars form a leading COLUMN prefix (so it can be probed
269// as a key); `constrained = false` is the binary fallback's freedom to join on
270// any shared key. The constrained/unconstrained peak gap is exactly the
271// order-loss the adversarial-chain test measures.
272// ---------------------------------------------------------------------------
273
274/// Best (lowest estimated peak) order over all leaders. Returns `(order,
275/// peak)`, or `None` if no complete connected (and, when constrained,
276/// prefix-key-joinable) order exists from any leader.
277fn best_greedy_peak(
278    ctx: &WcojDispatchCtx,
279    atom_vars: &[Vec<usize>],
280    cards: &[u64],
281    constrained: bool,
282) -> Option<(Vec<usize>, u64)> {
283    let n = atom_vars.len();
284    let mut best: Option<(Vec<usize>, u64)> = None;
285    for leader in 0..n {
286        if let Some((order, peak)) = greedy_from_leader(ctx, atom_vars, cards, constrained, leader)
287        {
288            if best.as_ref().is_none_or(|(_, bp)| peak < *bp) {
289                best = Some((order, peak));
290            }
291        }
292    }
293    best
294}
295
296/// Greedy left-deep order from a fixed leader: repeatedly append the
297/// connected (and, when constrained, prefix-key-joinable) not-yet-placed atom
298/// that minimizes the running intermediate estimate. Returns `(order, peak)`
299/// or `None` if some atom can never be appended under the constraints.
300fn greedy_from_leader(
301    ctx: &WcojDispatchCtx,
302    atom_vars: &[Vec<usize>],
303    cards: &[u64],
304    constrained: bool,
305    leader: usize,
306) -> Option<(Vec<usize>, u64)> {
307    let n = atom_vars.len();
308    let mut order = Vec::with_capacity(n);
309    let mut placed = vec![false; n];
310    let mut bound: std::collections::HashSet<usize> = std::collections::HashSet::new();
311    order.push(leader);
312    placed[leader] = true;
313    bound.extend(atom_vars[leader].iter().copied());
314    let mut running = cards[leader].max(1);
315    let mut peak = running;
316    for _ in 1..n {
317        let mut choice: Option<(usize, u64)> = None;
318        for a in 0..n {
319            if placed[a] || !is_addable(&atom_vars[a], &bound, constrained) {
320                continue;
321            }
322            let new_running = estimate_join_onto_prefix(ctx, atom_vars, cards, &order, running, a);
323            if choice.as_ref().is_none_or(|(_, c)| new_running < *c) {
324                choice = Some((a, new_running));
325            }
326        }
327        let (a, new_running) = choice?;
328        order.push(a);
329        placed[a] = true;
330        bound.extend(atom_vars[a].iter().copied());
331        running = new_running;
332        peak = peak.max(running);
333    }
334    Some((order, peak))
335}

337/// Estimated peak of a SPECIFIC processing order (used to score the current
338/// traversal order). Returns `None` if the order is not valid under the
339/// constraints (e.g. the traversal order is not prefix-key-joinable).
340fn eval_order_peak(
341    ctx: &WcojDispatchCtx,
342    atom_vars: &[Vec<usize>],
343    cards: &[u64],
344    order: &[usize],
345    constrained: bool,
346) -> Option<u64> {
347    let mut bound: std::collections::HashSet<usize> = std::collections::HashSet::new();
348    let mut running = 0u64;
349    let mut peak = 0u64;
350    for (step, &a) in order.iter().enumerate() {
351        if step == 0 {
352            running = cards[a].max(1);
353        } else {
354            if !is_addable(&atom_vars[a], &bound, constrained) {
355                return None;
356            }
357            running = estimate_join_onto_prefix(ctx, atom_vars, cards, &order[..step], running, a);
358        }
359        bound.extend(atom_vars[a].iter().copied());
360        peak = peak.max(running);
361    }
362    Some(peak)
363}

365/// Whether atom `a` (its per-column var ids) can extend a prefix that has
366/// already bound `bound`. Always requires connectivity (a shares ≥1 bound
367/// var); when `constrained`, the bound vars must additionally be a leading
368/// COLUMN prefix of `a` (FJ's probe-key rule).
369fn is_addable(
370    a_vars: &[usize],
371    bound: &std::collections::HashSet<usize>,
372    constrained: bool,
373) -> bool {
374    if !a_vars.iter().any(|v| bound.contains(v)) {
375        return false; // disconnected — never a Cartesian step
376    }
377    if constrained {
378        let split = a_vars.iter().take_while(|v| bound.contains(v)).count();
379        // need ≥1 bound key prefix, and no bound var AFTER an unbound one.
380        if split == 0 || a_vars[split..].iter().any(|v| bound.contains(v)) {
381            return false;
382        }
383    }
384    true
385}

387/// Estimated intermediate size after joining atom `a` onto the current
388/// left-deep prefix, via the most selective already-placed partner. Uses the
389/// independence model `running × (pairwise(p,a) / card(p))`.
390fn estimate_join_onto_prefix(
391    ctx: &WcojDispatchCtx,
392    atom_vars: &[Vec<usize>],
393    cards: &[u64],
394    placed: &[usize],
395    running: u64,
396    a: usize,
397) -> u64 {
398    let mut best_new = u64::MAX;
399    for &p in placed {
400        let (p_keys, a_keys) = shared_keys(&atom_vars[p], &atom_vars[a]);
401        if p_keys.is_empty() {
402            continue;
403        }
404        let pairwise = pairwise_estimate(ctx, cards, p, a, &p_keys, &a_keys);
405        let fan_out = pairwise as f64 / (cards[p].max(1)) as f64;
406        let new_running = ((running as f64) * fan_out).ceil().max(1.0);
407        let new_running = if new_running >= u64::MAX as f64 {
408            u64::MAX
409        } else {
410            new_running as u64
411        };
412        best_new = best_new.min(new_running);
413    }
414    // Caller guarantees `a` shares a bound var with some placed atom.
415    best_new.max(1)
416}

418/// Column positions of the join variables shared between `p` and `a`.
419fn shared_keys(p_vars: &[usize], a_vars: &[usize]) -> (Vec<usize>, Vec<usize>) {
420    let mut p_keys = Vec::new();
421    let mut a_keys = Vec::new();
422    for (pi, pv) in p_vars.iter().enumerate() {
423        if let Some(ai) = a_vars.iter().position(|av| av == pv) {
424            p_keys.push(pi);
425            a_keys.push(ai);
426        }
427    }
428    (p_keys, a_keys)
429}

431/// Pairwise join-cardinality estimate. Prefers the authoritative
432/// `StatsManager::estimate_join_cardinality` over the ACTUAL key columns when
433/// both relations have populated stats (cached selectivity / column distinct
434/// estimates then refine it); otherwise applies the same
435/// default model (10% selectivity) over the ground-truth buffer cardinalities,
436/// so statless single-rule joins (where scans never populate StatsManager) are
437/// still planned.
438fn pairwise_estimate(
439    ctx: &WcojDispatchCtx,
440    cards: &[u64],
441    p: usize,
442    a: usize,
443    p_keys: &[usize],
444    a_keys: &[usize],
445) -> u64 {
446    let stats_present = ctx
447        .stats
448        .get_relation_stats(ctx.slot_rels[p])
449        .is_some_and(|s| s.cardinality > 0)
450        && ctx
451            .stats
452            .get_relation_stats(ctx.slot_rels[a])
453            .is_some_and(|s| s.cardinality > 0);
454    if stats_present {
455        ctx.stats
456            .estimate_join_cardinality(ctx.slot_rels[p], ctx.slot_rels[a], p_keys, a_keys)
457    } else {
458        let prod = (cards[p] as u128) * (cards[a] as u128) / DEFAULT_JOIN_SELECTIVITY_DEN;
459        prod.min(u64::MAX as u128).max(1) as u64
460    }
461}

687    fn slots(n: usize) -> Vec<RelId> {
688        (0..n as u32).map(RelId).collect()

691    /// Adversarial chain q(A,E):-e1(A,B),e2(B,C),e3(C,D),e4(D,E): the ONLY
692    /// prefix-key-joinable order is left-to-right (which materializes the N²
693    /// prefix), while the binary plan can start from the 1-row tail. The
694    /// planner must DECLINE.
695    #[test]
696    fn order_planner_declines_unreorderable_blowup_chain() {
697        let cards = [100u64, 100, 10_000, 1];
698        let stats = stats_with_cards(&cards); // populated, but no cached selectivity
699        let rels = slots(4);
700        let ctx = nway_ctx(&stats, &rels);
701        // e1=[A,B] e2=[B,C] e3=[C,D] e4=[D,E] (vars A=0..E=4)
702        let atom_vars = vec![vec![0, 1], vec![1, 2], vec![2, 3], vec![3, 4]];
703        let model = CardinalityAwareCostModel::default();
704        assert!(
705            matches!(
706                model.plan_free_join_order(&ctx, &atom_vars, &cards),
707                FjOrderDecision::Decline
708            ),
709            "FJ-forced chain order blows up vs the tail-first binary plan → decline"
710        );
711    }

755    /// Alternating-cardinality 4-cycle: the traversal order starts on a big
756    /// relation (bad prefix), but a rotation starting on a 1-row relation is
757    /// both prefix-key-joinable AND competitive — the planner must REORDER to
758    /// a non-identity valid permutation.
759    #[test]
760    fn order_planner_reorders_alternating_cycle() {
761        let cards = [10_000u64, 1, 10_000, 1];
762        let stats = stats_with_cards(&cards);
763        let rels = slots(4);
764        let ctx = nway_ctx(&stats, &rels);
765        // r1=[A,B] r2=[B,C] r3=[C,D] r4=[D,A] (vars A=0,B=1,C=2,D=3)
766        let atom_vars = vec![vec![0, 1], vec![1, 2], vec![2, 3], vec![3, 0]];
767        let model = CardinalityAwareCostModel::default();
768        match model.plan_free_join_order(&ctx, &atom_vars, &cards) {
769            FjOrderDecision::Reorder(order) => {
770                let mut sorted = order.clone();
771                sorted.sort_unstable();
772                assert_eq!(sorted, vec![0, 1, 2, 3], "order must be a permutation");
773                assert_ne!(order, vec![0, 1, 2, 3], "must differ from traversal order");
774                assert!(
775                    eval_order_peak(&ctx, &atom_vars, &cards, &order, true).is_some(),
776                    "reordered plan must be prefix-key-joinable"
777                );
778            }
779            other => panic!("expected Reorder, got {:?}", DecisionDbg(&other)),
780        }
781    }

799    // Minimal Debug for panic messages (FjOrderDecision is not Debug).
800    struct DecisionDbg<'a>(&'a FjOrderDecision);
801    impl std::fmt::Debug for DecisionDbg<'_> {
802        fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
803            match self.0 {
804                FjOrderDecision::KeepDefault => write!(f, "KeepDefault"),
805                FjOrderDecision::Reorder(o) => write!(f, "Reorder({o:?})"),
806                FjOrderDecision::Decline => write!(f, "Decline"),
807            }
808        }
809    }
