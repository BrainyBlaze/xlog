1//! Hand-written fast path for simple ground facts inside `parse_program`.
2//!
3//! In the pest grammar a fact is the 13th `statement` alternative, so every
4//! `p(1, 2).` line is first tried as a function definition, a rule (whose head
5//! parses completely before `:-` fails), a probabilistic fact, ... and only then
6//! as a fact — and every token allocates a `Pair`. Fact-heavy programs (ILP
7//! event logs, EDB dumps) therefore spend most of their frontend time in the
8//! parser.
9//!
10//! This module scans the source once, recognises statements of the shape
11//!
12//! ```text
13//! ident "(" [ term { "," term } ] ")" "."
14//! ```
15//!
16//! where every `term` is an integer, a float, a symbol (ident), a variable, `_`
17//! or a `"string"`, builds the very same AST nodes `build_term`/`build_fact`
18//! would, and leaves *everything else* (rules, declarations, pragmas, facts with
19//! compound/list arguments, anything with a comment between tokens, ...) to pest
20//! as a "residual" source. The caller parses the residual with pest and merges
21//! the two statement streams back in source order.
22//!
23//! Soundness argument (why the residual cannot change what pest sees):
24//!
25//! * The residual is the original source with every recognised fact **blanked
26//!   out** — replaced byte for byte by spaces (newlines kept). It has the same
27//!   length, so byte offsets in the residual are byte offsets in the original
28//!   (no remapping), pest's error positions are the original ones, and the
29//!   tokens around a removed fact can never fuse (`X > 1.` `nn(1).` `0.5::c(1).`
30//!   stays `X > 1.` `      ` `0.5::c(1).` rather than becoming `1.0.5`).
31//!   Blanks are grammar WHITESPACE, which is implicit between statements.
32//! * A statement boundary is a `.` outside string literals and comments that is
33//!   not between two ASCII digits and is not part of the univ operator `=..`.
34//!   In the grammar `.` occurs only as a statement terminator, inside
35//!   `float_num`/`prob_num` (digits on both sides, atomic rules, no
36//!   whitespace), inside `string_lit` (`"..."`, no escapes), inside `//`
37//!   comments and in `=..` — the scanner handles each of these explicitly.
38//! * The fast path only accepts a *complete* fact at a statement start. Every
39//!   grammar alternative tried before `fact` either starts with a keyword that
40//!   cannot be followed by `(...)` `.` in a way that parses (`pred`, `domain`,
41//!   `use`, `func`, `nn`, `learnable` — all need more tokens), or is
42//!   `evidence(...)`/`query(...)`, which are excluded by name here. For the
43//!   remaining text pest itself would pick `fact`.
44//! * Symbols are interned when a fact is merged into the program, i.e. in
45//!   source order interleaved with pest's statements, so symbol ids are the
46//!   ones the reference parser would assign.
47//! * If the residual fails to parse or build for any reason, the caller re-runs
48//!   the pure-pest parser on the original source, so error messages (and their
49//!   positions) are exactly the reference ones.
50//!
51//! `crates/xlog-logic/tests/parse_fast_facts*.rs` check all of this against the
52//! reference parser on the in-repo corpus, adversarial snippets and randomised
53//! statement sequences.

57use crate::ast::{Atom, Rule as AstRule, Term};

59/// One fact recognised by the fast path, with its byte offset in the original
60/// source (used to merge with pest's statements in source order).
61#[derive(Debug)]
62pub(crate) struct FastFact<'a> {
63    /// Byte offset of the fact's first byte in the original source.
64    pub offset: usize,
65    /// The fact; symbol terms are placeholders until [`FastFact::into_rule`].
66    rule: AstRule,
67    /// `(term index, symbol text)` for every symbol term of the head, interned
68    /// only when the fact is merged so that interning order is source order.
69    symbols: Vec<(usize, &'a str)>,
70}
71
72impl FastFact<'_> {
73    /// Finish the fact: intern its symbols (now, in merge order) and return it.
74    pub(crate) fn into_rule(self) -> AstRule {
75        let mut rule = self.rule;
76        for (index, text) in self.symbols {
77            rule.head.terms[index] = Term::Symbol(symbol::intern(text));
78        }
79        rule
80    }
81}

83/// Result of scanning a source text.
84#[derive(Debug, Default)]
85pub(crate) struct FactScan<'a> {
86    /// Facts handled here, in source order.
87    pub facts: Vec<FastFact<'a>>,
88    /// The source with those facts blanked out, for pest. Same length as the
89    /// source, so offsets coincide. Empty when no fact was recognised.
90    pub residual: String,
91    /// Bytes occupied by the recognised facts (what pest does not re-parse).
92    pub fact_bytes: usize,
93}

95/// Statement-start predicates the grammar treats specially (they come before
96/// `fact` in `statement` and are fact-shaped): never fast-path them.
97fn is_reserved_statement_head(ident: &str) -> bool {
98    matches!(ident, "evidence" | "query")
99}
100
101#[inline]
102fn is_ws(b: u8) -> bool {
103    // Exactly the grammar's WHITESPACE rule: " " | "\t" | "\r" | "\n".
104    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
105}
106
107#[inline]
108fn is_ident_start(b: u8) -> bool {
109    b.is_ascii_lowercase()
110}
111
112#[inline]
113fn is_ident_continue(b: u8) -> bool {
114    b.is_ascii_alphanumeric() || b == b'_'
115}

117/// Skip whitespace and `//` comments (what pest skips implicitly between
118/// statements). Returns the offset of the next significant byte.
119fn skip_ws_and_comments(src: &[u8], mut pos: usize) -> usize {
120    loop {
121        while pos < src.len() && is_ws(src[pos]) {
122            pos += 1;
123        }
124        if pos + 1 < src.len() && src[pos] == b'/' && src[pos + 1] == b'/' {
125            while pos < src.len() && src[pos] != b'\n' {
126                pos += 1;
127            }
128            continue;
129        }
130        return pos;
131    }
132}
133
134/// Skip whitespace only (inside a fact we bail on comments instead of
135/// modelling pest's implicit COMMENT between tokens).
136fn skip_ws(src: &[u8], mut pos: usize) -> usize {
137    while pos < src.len() && is_ws(src[pos]) {
138        pos += 1;
139    }
140    pos
141}

143/// From `pos` (a statement start that is not a fast fact), find the offset just
144/// past the statement terminator `.`; `src.len()` if there is none.
145fn skip_statement(src: &[u8], mut pos: usize) -> usize {
146    let mut in_string = false;
147    while pos < src.len() {
148        let b = src[pos];
149        if in_string {
150            if b == b'"' {
151                in_string = false;
152            }
153            pos += 1;
154            continue;
155        }
156        match b {
157            b'"' => in_string = true,
158            // The univ operator `=..` is the one token besides float/prob
159            // literals that contains `.`: neither dot is a terminator.
160            b'=' if pos + 2 < src.len() && src[pos + 1] == b'.' && src[pos + 2] == b'.' => {
161                pos += 3;
162                continue;
163            }
164            b'/' if pos + 1 < src.len() && src[pos + 1] == b'/' => {
165                while pos < src.len() && src[pos] != b'\n' {
166                    pos += 1;
167                }
168                continue;
169            }
170            b'.' => {
171                let decimal = pos > 0
172                    && src[pos - 1].is_ascii_digit()
173                    && pos + 1 < src.len()
174                    && src[pos + 1].is_ascii_digit();
175                if !decimal {
176                    return pos + 1;
177                }
178            }
179            _ => {}
180        }
181        pos += 1;
182    }
183    src.len()
184}

186/// Try to recognise a simple ground fact starting exactly at `start`.
187/// Returns the fact and the offset just past its terminating `.`.
188fn try_fact(source: &str, start: usize) -> Option<(FastFact<'_>, usize)> {
189    let src = source.as_bytes();
190    let mut pos = start;
191    if pos >= src.len() || !is_ident_start(src[pos]) {
192        return None;
193    }
194    while pos < src.len() && is_ident_continue(src[pos]) {
195        pos += 1;
196    }
197    let predicate = &source[start..pos];
198    if is_reserved_statement_head(predicate) {
199        return None;
200    }
201    pos = skip_ws(src, pos);
202    if pos >= src.len() || src[pos] != b'(' {
203        return None;
204    }
205    pos += 1;
206    let mut terms = Vec::new();
207    let mut symbols = Vec::new();
208    pos = skip_ws(src, pos);
209    if pos < src.len() && src[pos] == b')' {
210        pos += 1;
211    } else {
212        loop {
213            let (term, next) = try_term(source, pos)?;
214            match term {
215                FastTerm::Term(term) => terms.push(term),
216                FastTerm::Symbol(text) => {
217                    symbols.push((terms.len(), text));
218                    terms.push(Term::Symbol(0));
219                }
220            }
221            pos = skip_ws(src, next);
222            match src.get(pos) {
223                Some(b',') => {
224                    pos = skip_ws(src, pos + 1);
225                }
226                Some(b')') => {
227                    pos += 1;
228                    break;
229                }
230                _ => return None,
231            }
232        }
233    }
234    pos = skip_ws(src, pos);
235    if pos >= src.len() || src[pos] != b'.' {
236        return None;
237    }
238    let fact = FastFact {
239        offset: start,
240        rule: AstRule {
241            head: Atom {
242                predicate: predicate.to_string(),
243                terms,
244            },
245            body: vec![],
246        },
247        symbols,
248    };
249    Some((fact, pos + 1))
250}

252/// A recognised term: either final, or a symbol whose interning is deferred.
253enum FastTerm<'a> {
254    Term(Term),
255    Symbol(&'a str),
256}
257
258/// Recognise one simple term at `pos`: variable, `_`, integer, float, string
259/// literal or symbol. Returns the term and the offset just past it. Anything
260/// else (compound, list, cons, comment, ...) returns `None`.
261fn try_term(source: &str, pos: usize) -> Option<(FastTerm<'_>, usize)> {
262    let src = source.as_bytes();
263    let b = *src.get(pos)?;
264    if b == b'_' {
265        // `anonymous` is exactly "_"; the caller then requires `,` or `)`,
266        // which rejects `_foo` just like the grammar does.
267        return Some((FastTerm::Term(Term::Anonymous), pos + 1));
268    }
269    if b.is_ascii_uppercase() {
270        let mut end = pos + 1;
271        while end < src.len() && is_ident_continue(src[end]) {
272            end += 1;
273        }
274        return Some((
275            FastTerm::Term(Term::Variable(source[pos..end].to_string())),
276            end,
277        ));
278    }
279    if b == b'-' || b.is_ascii_digit() {
280        let mut end = pos;
281        if src[end] == b'-' {
282            end += 1;
283        }
284        let digits_start = end;
285        while end < src.len() && src[end].is_ascii_digit() {
286            end += 1;
287        }
288        if end == digits_start {
289            return None;
290        }
291        // float_num = "-"? digits "." digits — atomic, digits on both sides.
292        if end + 1 < src.len() && src[end] == b'.' && src[end + 1].is_ascii_digit() {
293            end += 1;
294            while end < src.len() && src[end].is_ascii_digit() {
295                end += 1;
296            }
297            let val: f64 = source[pos..end].parse().ok()?;
298            return Some((FastTerm::Term(Term::Float(val)), end));
299        }
300        let val: i64 = source[pos..end].parse().ok()?;
301        return Some((FastTerm::Term(Term::Integer(val)), end));
302    }
303    if b == b'"' {
304        let rel = src[pos + 1..].iter().position(|&c| c == b'"')?;
305        let end = pos + 1 + rel;
306        return Some((
307            FastTerm::Term(Term::String(source[pos + 1..end].to_string())),
308            end + 1,
309        ));
310    }
311    if is_ident_start(b) {
312        let mut end = pos + 1;
313        while end < src.len() && is_ident_continue(src[end]) {
314            end += 1;
315        }
316        // `ident "("` would be a compound term: leave it to pest.
317        let after = skip_ws(src, end);
318        if after < src.len() && src[after] == b'(' {
319            return None;
320        }
321        return Some((FastTerm::Symbol(&source[pos..end]), end));
322    }
323    None
324}

326/// Append `source[start..end]` to the residual with every byte blanked to a
327/// space, newlines kept (line numbers in pest diagnostics stay intact).
328fn push_blanked(residual: &mut String, source: &str, start: usize, end: usize) {
329    const SPACES: &str = "                                                                ";
330    for line in source[start..end].split_inclusive('\n') {
331        let (body, newline) = match line.strip_suffix('\n') {
332            Some(body) => (body, true),
333            None => (line, false),
334        };
335        let mut remaining = body.len();
336        while remaining > 0 {
337            let chunk = remaining.min(SPACES.len());
338            residual.push_str(&SPACES[..chunk]);
339            remaining -= chunk;
340        }
341        if newline {
342            residual.push('\n');
343        }
344    }
345}

347/// Scan `source`, extracting simple ground facts and producing the residual.
348pub(crate) fn scan(source: &str) -> FactScan<'_> {
349    let src = source.as_bytes();
350    let mut out = FactScan::default();
351    let mut pos = 0usize;
352    // Start of the source run not yet copied into the residual. The residual
353    // is only materialised once the first fact is found, so programs without
354    // simple facts pay for the scan but not for a copy of their source.
355    let mut copied_to = 0usize;
356    while pos < src.len() {
357        let stmt_start = skip_ws_and_comments(src, pos);
358        if stmt_start >= src.len() {
359            break;
360        }
361        if let Some((fact, end)) = try_fact(source, stmt_start) {
362            if out.facts.is_empty() {
363                out.residual.reserve(source.len());
364            }
365            out.residual.push_str(&source[copied_to..stmt_start]);
366            push_blanked(&mut out.residual, source, stmt_start, end);
367            out.fact_bytes += end - stmt_start;
368            out.facts.push(fact);
369            copied_to = end;
370            pos = end;
371        } else {
372            pos = skip_statement(src, stmt_start);
373        }
374    }
375    if !out.facts.is_empty() {
376        out.residual.push_str(&source[copied_to..]);
377        debug_assert_eq!(out.residual.len(), source.len());
378    }
379    out
380}

383mod tests {

386    #[test]
387    fn recognises_simple_facts_and_leaves_the_rest() {
388        let src = "p(1, -2, 3.5, \"s\", sym, X, _).\nq(X) :- p(X, _, _, _, _, _, _).\nr().\n";
389        let scan = scan(src);
390        assert_eq!(scan.facts.len(), 2);
391        let facts: Vec<AstRule> = scan.facts.into_iter().map(FastFact::into_rule).collect();
392        assert_eq!(facts[0].head.predicate, "p");
393        assert_eq!(
394            facts[0].head.terms,
395            vec![
396                Term::Integer(1),
397                Term::Integer(-2),
398                Term::Float(3.5),
399                Term::String("s".to_string()),
400                Term::Symbol(symbol::intern("sym")),
401                Term::Variable("X".to_string()),
402                Term::Anonymous,
403            ]
404        );
405        assert_eq!(facts[1].head.predicate, "r");
406        assert_eq!(scan.residual.len(), src.len());
407        assert_eq!(scan.residual.trim(), "q(X) :- p(X, _, _, _, _, _, _).");
408    }

410    #[test]
411    fn leaves_compound_list_comment_and_reserved_to_pest() {
412        for src in [
413            "p(f(1)).",
414            "p([1]).",
415            "p( // c\n 1).",
416            "evidence(foo(1), true).",
417            "query(x).",
418            "p(1) :- q(1).",
419            "p(1)",
420            "p(_foo).",
421            "p(1.).",
422        ] {
423            let scan = scan(src);
424            assert!(scan.facts.is_empty(), "{src:?} should not be fast-pathed");
425            assert!(
426                scan.residual.is_empty(),
427                "no residual is built without facts"
428            );
429        }
430    }

432    #[test]
433    fn statement_skipping_respects_strings_comments_decimals_and_univ() {
434        let src = "x(X) :- X > 1.5, y(\"a.b\"). // c.\np(1).";
435        let first = scan(src);
436        assert_eq!(first.facts.len(), 1);
437        assert_eq!(first.facts[0].offset, src.find("p(1)").unwrap());
438
439        // `=..` is one token: its dots are not terminators, so `f(1)` below is
440        // the univ's right-hand side, not a fact.
441        let src = "r(X) :- X =.. f(1).\nq(g(1)).";
442        let second = scan(src);
443        assert!(second.facts.is_empty(), "{:?}", second.facts);
444    }

446    #[test]
447    fn blanking_keeps_offsets_and_never_fuses_neighbours() {
448        let src = "u(X) :- X > 1.nn(1).0.5::c(1).";
449        let scan = scan(src);
450        assert_eq!(scan.facts.len(), 1);
451        assert_eq!(scan.residual.len(), src.len());
452        assert_eq!(scan.residual, "u(X) :- X > 1.      0.5::c(1).");
453        assert_eq!(scan.fact_bytes, "nn(1).".len());
