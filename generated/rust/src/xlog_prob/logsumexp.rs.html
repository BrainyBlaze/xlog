3fn nan_error(context: &str) -> XlogError {
4    XlogError::Compilation(format!("{context} encountered NaN"))
5}

7/// Validates one true/false log-weight pair for value-only circuit evaluation.
8///
9/// Both polarities are part of the weight input contract even when the circuit consumes only one
10/// of them. Positive and negative infinity remain valid for value-only evaluation.
11pub(crate) fn validate_circuit_log_weight_pair(weights: (f64, f64)) -> Result<(f64, f64)> {
12    if weights.0.is_nan() || weights.1.is_nan() {
13        return Err(nan_error("circuit log weights"));
14    }
15    Ok(weights)
16}

18/// Rejects a NaN produced while evaluating a circuit node or read back from a GPU root.
19pub(crate) fn validate_circuit_value(value: f64) -> Result<f64> {
20    if value.is_nan() {
21        return Err(nan_error("circuit evaluation"));
22    }
23    Ok(value)
24}

26/// Validates gradient arrays returned across a device-to-host boundary.
27///
28/// Gradient outputs must be finite even when value-only circuit evaluation legitimately returns
29/// infinity. True-polarity gradients are scanned before false-polarity gradients, and each array
30/// is scanned in ascending variable order, so the reported error is deterministic.
31pub(crate) fn validate_circuit_gradient_values(
32    grad_true: &[f64],
33    grad_false: &[f64],
34) -> Result<()> {
35    for (polarity, gradients) in [("true", grad_true), ("false", grad_false)] {
36        if let Some((index, _)) = gradients
37            .iter()
38            .enumerate()
39            .find(|(_, gradient)| !gradient.is_finite())
40        {
41            return Err(XlogError::Compilation(format!(
42                "circuit gradient {polarity}[{index}] is non-finite"
43            )));
44        }
45    }
46    Ok(())
47}

49/// Computes circuit log-sum-exp with a stable max shift for finite inputs.
50///
51/// Empty inputs and inputs containing only negative infinity return negative infinity.
52/// Positive infinity returns positive infinity when no NaN is present. The initial full
53/// scan gives NaN explicit precedence over infinity and returns a typed
54/// [`XlogError::Compilation`] error whenever any input is NaN.
55pub(crate) fn circuit_logsumexp(values: &[f64]) -> Result<f64> {
56    let mut max = f64::NEG_INFINITY;
57    for &value in values {
58        if value.is_nan() {
59            return Err(nan_error("circuit log-sum-exp"));
60        }
61        if value > max {
62            max = value;
63        }
64    }
65    if max.is_infinite() {
66        return Ok(max);
67    }
68
69    let mut sum = 0.0;
70    for &value in values {
71        sum += (value - max).exp();
72    }
73    Ok(max + sum.ln())
74}

81    #[test]
82    fn circuit_logsumexp_edge_contract() {
83        let valid_cases: &[(&str, &[f64], f64)] = &[
84            ("empty", &[], f64::NEG_INFINITY),
85            (
86                "all negative infinity",
87                &[f64::NEG_INFINITY, f64::NEG_INFINITY],
88                f64::NEG_INFINITY,
89            ),
90            (
91                "positive infinity",
92                &[f64::NEG_INFINITY, 3.0, f64::INFINITY],
93                f64::INFINITY,
94            ),
95            (
96                "large positive finite",
97                &[1000.0, 999.0],
98                1000.3132616875182,
99            ),
100            (
101                "large negative finite",
102                &[-1000.0, -1001.0],
103                -999.6867383124818,
104            ),
105        ];
106
107        for &(name, values, expected) in valid_cases {
108            let actual = circuit_logsumexp(values)
109                .unwrap_or_else(|error| panic!("{name}: expected a value, got {error}"));
110            if expected.is_infinite() {
111                assert_eq!(actual, expected, "{name}");
112            } else {
113                assert!((actual - expected).abs() < 1e-12, "{name}: {actual}");
114            }
115        }
116
117        for values in [
118            &[f64::NAN][..],
119            &[0.0, f64::NAN][..],
120            &[f64::INFINITY, f64::NAN][..],
121        ] {
122            let error = circuit_logsumexp(values).expect_err("NaN must be rejected");
123            assert!(
124                matches!(error, XlogError::Compilation(ref message) if message.contains("NaN")),
125                "unexpected error: {error}"
126            );
127        }
128    }
129
130    #[test]
