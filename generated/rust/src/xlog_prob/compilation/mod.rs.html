47/// Per-stage compilation timing (populated only when XLOG_WARMUP_PROFILE=1).
48#[derive(Debug, Clone, Default)]
49pub struct CircuitCompileProfile {
50    /// Seconds spent computing the canonical CNF hash.
51    pub cnf_hash_sec: f64,
52    /// Seconds spent compiling the circuit.
53    pub d4_compile_sec: f64,
54    /// Seconds spent verifying logical equivalence.
55    pub verify_sec: f64,
56    /// Seconds spent smoothing the final circuit.
57    pub smooth_sec: f64,
58    /// Seconds spent storing the verified circuit in cache.
59    pub cache_store_sec: f64,
60    /// Seconds spent constructing the free-variable mask.
61    pub free_var_mask_sec: f64,
62    /// Whether the circuit came from the resident GPU cache.
63    pub gpu_cache_hit: bool,
64    /// Whether the circuit came from the verified disk cache.
65    pub disk_cache_hit: bool,
66    /// BFS frontier item count after `frontier_depth` expansion steps
67    /// (0 on cache hits or when profiling is disabled).
68    pub frontier_items: u32,
69}

71/// Monotonic event ledger for one exact-circuit materialization path.
72///
73/// Mutation is private to this module so counts can only change at the GPU
74/// compiler, verified disk-restore, GPU-cache-hit, and successful
75/// materialization boundaries below. The exact state retains this ledger and
76/// snapshots it around every prepared evaluation.
77#[derive(Debug, Default)]
78pub(crate) struct CircuitCompilationLedger {
79    compiler_invocations: AtomicU64,
80    materializations: AtomicU64,
81    disk_cache_restores: AtomicU64,
82    gpu_cache_hits: AtomicU64,
83}

85#[cfg(feature = "host-io")]
86#[derive(Debug, Clone, Copy, PartialEq, Eq)]
87pub(crate) struct CircuitCompilationSnapshot {
88    pub(crate) compiler_invocations: u64,
89    pub(crate) materializations: u64,
90    pub(crate) disk_cache_restores: u64,
91    pub(crate) gpu_cache_hits: u64,
92}
93
94impl CircuitCompilationLedger {
95    pub(crate) fn new() -> Self {
96        Self::default()
97    }
98
99    #[cfg(feature = "host-io")]
100    pub(crate) fn snapshot(&self) -> CircuitCompilationSnapshot {
101        CircuitCompilationSnapshot {
102            compiler_invocations: self.compiler_invocations.load(Ordering::Acquire),
103            materializations: self.materializations.load(Ordering::Acquire),
104            disk_cache_restores: self.disk_cache_restores.load(Ordering::Acquire),
105            gpu_cache_hits: self.gpu_cache_hits.load(Ordering::Acquire),
106        }
107    }
108
109    fn record_compiler_invocation(&self) -> Result<()> {
110        increment_circuit_event(
111            &self.compiler_invocations,
112            "GPU circuit compiler invocation",
113        )
114    }
115
116    fn record_materialization(&self) -> Result<()> {
117        increment_circuit_event(&self.materializations, "exact circuit materialization")
118    }
119
120    fn record_disk_cache_restore(&self) -> Result<()> {
121        increment_circuit_event(&self.disk_cache_restores, "verified disk-cache restoration")
122    }
123
124    fn record_gpu_cache_hit(&self) -> Result<()> {
125        increment_circuit_event(&self.gpu_cache_hits, "GPU circuit-cache hit")
126    }
127}

135fn increment_circuit_event(counter: &AtomicU64, event: &str) -> Result<()> {
136    counter
137        .fetch_update(Ordering::AcqRel, Ordering::Acquire, |value| {
138            value.checked_add(1)
139        })
140        .map(|_| ())
141        .map_err(|_| XlogError::Compilation(format!("{event} counter overflowed")))
142}

154/// Circuit whose exact topology has passed equivalence verification.
155///
156/// The field is private so topology-changing code cannot construct this type
157/// without going through [`certify_final_circuit`].
158struct CertifiedGpuCircuit(GpuXgcf);
159
160impl CertifiedGpuCircuit {
161    fn as_xgcf(&self) -> &GpuXgcf {
162        &self.0
163    }
164}

166impl DeviceRandomVarList {
167    /// Constructs a validated list from an existing device allocation and logical count.
168    pub fn from_device(list: TrackedCudaSlice<u32>, count: u32) -> Result<Self> {
169        let len = u32::try_from(list.len()).map_err(|_| {
170            XlogError::Compilation("DeviceRandomVarList: list length exceeds u32".to_string())
171        })?;
172        if count > len {
173            return Err(XlogError::Compilation(format!(
174                "DeviceRandomVarList: count {} exceeds list len {}",
175                count, len
176            )));
177        }
178        Ok(Self { list, count })
179    }
180
181    /// Uploads a host list of random-variable identifiers.
182    pub fn from_host(provider: &CudaKernelProvider, host: &[u32]) -> Result<Self> {
183        let memory = provider.memory();
184        let mut list = memory.alloc::<u32>(host.len())?;
185        if !host.is_empty() {
186            provider
187                .htod_sync_copy_into_tracked(host, &mut list)
188                .map_err(|e| {
189                    XlogError::Kernel(format!("DeviceRandomVarList upload failed: {}", e))
190                })?;
191        }
192        let count = u32::try_from(host.len()).map_err(|_| {
193            XlogError::Compilation("DeviceRandomVarList: host len exceeds u32".to_string())
194        })?;
195        Ok(Self { list, count })
196    }
197
198    /// Returns whether the logical list is empty.
199    pub fn is_empty(&self) -> bool {
200        self.count == 0
201    }
202
203    /// Returns the logical number of random variables.
204    pub fn count(&self) -> u32 {
205        self.count
206    }
207
208    /// Returns the device-resident variable identifiers.
209    pub fn list(&self) -> &TrackedCudaSlice<u32> {
210        &self.list
211    }
212}

226fn upload_disk_artifact_for_verification(
227    artifact: &disk_cache::CircuitArtifact,
228    provider: &Arc<CudaKernelProvider>,
229) -> Result<GpuXgcf> {
230    let memory = provider.memory().clone();
231
232    macro_rules! upload {
233        ($field:expr, $ty:ty, $name:literal) => {{
234            let mut device = memory.alloc::<$ty>($field.len())?;
235            provider
236                .htod_sync_copy_into_tracked($field, &mut device)
237                .map_err(|e| {
238                    XlogError::Kernel(format!("disk cache verify upload {} failed: {}", $name, e))
239                })?;
240            device
241        }};
242    }
243
244    let builder = GpuCircuitBuilder {
245        node_type: upload!(&artifact.node_type, u8, "node_type"),
246        child_offsets: upload!(&artifact.child_offsets, u32, "child_offsets"),
247        child_indices: upload!(&artifact.child_indices, u32, "child_indices"),
248        lit: upload!(&artifact.lit, i32, "lit"),
249        decision_var: upload!(&artifact.decision_var, u32, "decision_var"),
250        decision_child_false: upload!(&artifact.decision_child_false, u32, "decision_child_false"),
251        decision_child_true: upload!(&artifact.decision_child_true, u32, "decision_child_true"),
252    };
253    let layout = GpuCircuitLayout {
254        num_nodes: artifact.num_nodes,
255        num_edges: artifact.num_edges,
256        num_levels: artifact.num_levels,
257        level_offsets: upload!(&artifact.level_offsets, u32, "level_offsets"),
258        level_nodes: upload!(&artifact.level_nodes, u32, "level_nodes"),
259        root: artifact.root,
260        max_var: artifact.max_var,
261        num_nodes_device: None,
262        num_edges_device: None,
263    };
264
265    GpuXgcf::from_device(builder, layout, provider)
266}

268/// Compile CNF on GPU, then verify equivalence with GPU CDCL.
269pub fn compile_gpu_d4_and_verify(
270    cnf: &GpuCnf,
271    decision_var_limit: &TrackedCudaSlice<u32>,
272    provider: &Arc<CudaKernelProvider>,
273    config: &GpuCompileConfig,
274) -> Result<GpuXgcf> {
275    if config.cdcl_conflict_budget.is_some() {
276        return Err(XlogError::Compilation(
277            "cdcl_conflict_budget is not supported by the GPU CDCL verifier".to_string(),
278        ));
279    }
280    // Size guard BEFORE compile — the D4 compile itself can crash with a
281    // context-poisoning launch failure on a large CNF, earlier than the verify.
282    validation::check_verify_size_bound(cnf, "compile_gpu_d4_and_verify")?;
283    let circuit = gpu_d4::compile_gpu_d4(cnf, provider, config)?;
284    let cdcl = cdcl_config_from_compile(config)?;
285    validate_equivalence_gpu(
286        cnf,
287        decision_var_limit,
288        &circuit,
289        provider,
290        GpuEquivalenceConfig {
291            cdcl,
292            reuse_workspace: config.incremental_verify,
293        },
294    )?;
295    Ok(circuit)
296}

298/// Compile CNF on GPU, cache the circuit, then verify equivalence with GPU CDCL.
299///
300/// `canonical_cnf_hash`: a process-independent hash of the PIR structure, used as
301/// the `cnf_hash` in the disk cache key. Computed via [`crate::cnf::canonical_pir_hash`].
302/// If `None`, disk caching is skipped.
303pub fn compile_gpu_d4_and_verify_cached(
304    cnf: &GpuCnf,
305    decision_var_limit: &TrackedCudaSlice<u32>,
306    provider: &Arc<CudaKernelProvider>,
307    config: &GpuCompileConfig,
308    cache: &mut GpuCircuitCache,
309    random_vars: &DeviceRandomVarList,
310    canonical_cnf_hash: Option<u64>,
311) -> Result<(GpuCircuitCacheHandle, Option<CircuitCompileProfile>)> {
312    let ledger = CircuitCompilationLedger::new();
313    compile_gpu_d4_and_verify_cached_with_ledger(
314        cnf,
315        decision_var_limit,
316        provider,
317        config,
318        cache,
319        random_vars,
320        CircuitCompilationContext {
321            canonical_cnf_hash,
322            ledger: &ledger,
323        },
324    )
325}

327pub(crate) fn compile_gpu_d4_and_verify_cached_with_ledger(
328    cnf: &GpuCnf,
329    decision_var_limit: &TrackedCudaSlice<u32>,
330    provider: &Arc<CudaKernelProvider>,
331    config: &GpuCompileConfig,
332    cache: &mut GpuCircuitCache,
333    random_vars: &DeviceRandomVarList,
334    context: CircuitCompilationContext<'_>,
335) -> Result<(GpuCircuitCacheHandle, Option<CircuitCompileProfile>)> {
336    let CircuitCompilationContext {
337        canonical_cnf_hash,
338        ledger,
339    } = context;
340    if config.cdcl_conflict_budget.is_some() {
341        return Err(XlogError::Compilation(
342            "cdcl_conflict_budget is not supported by the GPU CDCL verifier".to_string(),
343        ));
344    }
345    // Size guard BEFORE compile (the D4 compile can crash earlier than
346    // the verify on a large CNF).
347    validation::check_verify_size_bound(cnf, "compile_gpu_d4_and_verify_cached")?;
348
349    let profiling = warmup_profiling_enabled()?;
350    let mut profile = CircuitCompileProfile::default();
351
352    // --- CNF hash stage ---
353    #[cfg(debug_assertions)]
354    eprintln!("[xlog-prob] compile_gpu_d4_and_verify_cached: hash_cnf_gpu");
355    let t_hash = if profiling {
356        Some(Instant::now())
357    } else {
358        None
359    };
360    let key = gpu_cache::hash_cnf_gpu(cnf, provider)?;
361    if let Some(t0) = t_hash {
362        provider
363            .device()
364            .synchronize()
365            .map_err(|e| XlogError::Kernel(format!("sync after hash_cnf_gpu: {}", e)))?;
366        profile.cnf_hash_sec = t0.elapsed().as_secs_f64();
367    }
368    #[cfg(debug_assertions)]
369    {
370        if !profiling {
371            provider
372                .device()
373                .synchronize()
374                .map_err(|e| XlogError::Kernel(format!("sync after hash_cnf_gpu failed: {}", e)))?;
375        }
376    }
377    #[cfg(debug_assertions)]
378    eprintln!("[xlog-prob] compile_gpu_d4_and_verify_cached: lookup_or_insert_device");
379    let lookup = cache.lookup_or_insert_device(&key)?;
380    let mut handle = lookup.into_handle()?;
381
382    // --- Disk cache check (only on GPU cache miss) ---
383    //
384    // D→H copy compile_needed to decide whether we need to compile at all.
385    // If compile_needed == 0, the GPU cache already has the circuit (GPU cache hit).
386    // If compile_needed == 1, we check the disk cache before falling through to the
387    // GPU-native Decision-DNNF compiler.
388    let compile_needed_host: Vec<u32> = provider
389        .device()
390        .inner()
391        .dtoh_sync_copy(handle.compile_needed_device())
392        .map_err(|e| XlogError::Kernel(format!("dtoh compile_needed: {}", e)))?;
393    let compile_needed = compile_needed_host[0];
394
395    // GPU cache hit — short-circuit the entire compile pipeline.
396    if compile_needed == 0 {
397        profile.gpu_cache_hit = true;
398        ledger.record_gpu_cache_hit()?;
399        ledger.record_materialization()?;
400        let out_profile = if profiling { Some(profile) } else { None };
401        return Ok((handle, out_profile));
402    }
403
404    // Build the disk cache key (we know compile_needed == 1 at this point).
405    // The canonical PIR hash keeps semantically stable cache identity, while the encoded CNF
406    // hash guards symmetric conditioned programs whose PIR shape is identical but whose CNF is not.
407    let cache_key = if compile_needed == 1 {
408        if let Some(canonical_hash) = canonical_cnf_hash {
409            let config_hash = hash_compile_config(config);
410            let random_vars_hash = hash_random_vars(random_vars, provider)?;
411            let gpu_cnf_hash = read_gpu_cnf_hash(&key, provider)?;
412            let cnf_hash = combine_disk_cnf_hash(canonical_hash, gpu_cnf_hash);
413            let sm = detect_compute_capability(provider)?;
414            Some(disk_cache::CircuitCacheKey {
415                cnf_hash,
416                config_hash,
417                random_vars_hash,
418                sm,
419            })
420        } else {
421            None
422        }
423    } else {
424        None
425    };
426
427    // Check disk cache on GPU cache miss
428    if let Some(ref disk_key) = cache_key {
429        #[cfg(debug_assertions)]
430        eprintln!("[xlog-prob] compile_gpu_d4_and_verify_cached: checking disk cache");
431        if let Ok(Some(artifact)) = disk_cache::read_artifact(disk_key) {
432            #[cfg(debug_assertions)]
433            eprintln!("[xlog-prob] compile_gpu_d4_and_verify_cached: disk cache hit");
434            // Stale-artifact guard: the canonical PIR hash keeps semantically-stable
435            // cache identity, so an entry written by an earlier engine whose ENCODING
436            // differed (different var universe) can be served for the same key. A
437            // cached circuit referencing vars beyond the freshly-encoded CNF is
438            // staleness, not an engine bug — evict and fall through to a fresh
439            // compile (which overwrites the entry). Fail-closed equivalence checking
440            // stays authoritative for freshly-compiled circuits below.
441            if artifact.max_var > cnf.var_cap {
442                #[cfg(debug_assertions)]
443                eprintln!(
444                    "[xlog-prob] compile_gpu_d4_and_verify_cached: stale disk artifact \
445                     (max_var {} > cnf var_cap {}), evicting",
446                    artifact.max_var, cnf.var_cap
447                );
448                disk_cache::evict_artifact(disk_key);
449            } else {
450                let circuit = upload_disk_artifact_for_verification(&artifact, provider)?;
451                let verifier_decision_var_limit = if random_vars.is_empty() {
452                    &cnf.num_vars
453                } else {
454                    decision_var_limit
455                };
456                let cdcl = cdcl_config_from_compile(config)?;
457                let t_verify = if profiling {
458                    Some(Instant::now())
459                } else {
460                    None
461                };
462                match certify_final_circuit(
463                    cnf,
464                    verifier_decision_var_limit,
465                    circuit,
466                    provider,
467                    GpuEquivalenceConfig {
468                        cdcl,
469                        reuse_workspace: config.incremental_verify,
470                    },
471                    handle.compile_needed_device(),
472                ) {
473                    Ok(_certified_circuit) => {
474                        if let Some(t0) = t_verify {
475                            provider.device().synchronize().map_err(|e| {
476                                XlogError::Kernel(format!("sync after disk cache verify: {}", e))
477                            })?;
478                            profile.verify_sec = t0.elapsed().as_secs_f64();
479                        }
480                        cache.restore_from_host_arrays(&mut handle, &artifact)?;
481                        provider.device().synchronize().map_err(|e| {
482                            XlogError::Kernel(format!("sync after disk cache restore: {}", e))
483                        })?;
484                        profile.disk_cache_hit = true;
485                        ledger.record_disk_cache_restore()?;
486                        ledger.record_materialization()?;
487                        let out_profile = if profiling { Some(profile) } else { None };
488                        return Ok((handle, out_profile));
489                    }
490                    Err(_stale) => {
491                        // A cached artifact failing equivalence against the current
492                        // CNF is a stale entry (fresh-compile verification below
493                        // remains fail-closed). Evict and recompile.
494                        #[cfg(debug_assertions)]
495                        eprintln!(
496                            "[xlog-prob] compile_gpu_d4_and_verify_cached: cached circuit \
497                             failed equivalence against current CNF ({_stale}), evicting"
498                        );
499                        disk_cache::evict_artifact(disk_key);
500                    }
501                }
502            }
503        }
504        #[cfg(debug_assertions)]
505        eprintln!("[xlog-prob] compile_gpu_d4_and_verify_cached: disk cache miss");
506    }
507
508    let d4_config = d4_config_for_smoothing(config, random_vars.count())?;
509
510    // --- GPU-native Decision-DNNF compile stage ---
511    #[cfg(debug_assertions)]
512    eprintln!("[xlog-prob] compile_gpu_d4_and_verify_cached: compile_gpu_d4_gated");
513    let t_d4 = if profiling {
514        Some(Instant::now())
515    } else {
516        None
517    };
518    ledger.record_compiler_invocation()?;
519    let (circuit_base, frontier_items) = gpu_d4::compile_gpu_d4_gated_with_stats(
520        cnf,
521        provider,
522        &d4_config,
523        handle.compile_needed_device(),
524    )?;
525    profile.frontier_items = frontier_items;
526    if let Some(t0) = t_d4 {
527        provider
528            .device()
529            .synchronize()
530            .map_err(|e| XlogError::Kernel(format!("sync after d4 compile: {}", e)))?;
531        profile.d4_compile_sec = t0.elapsed().as_secs_f64();
532    }
533    #[cfg(debug_assertions)]
534    {
535        if !profiling {
536            provider.device().synchronize().map_err(|e| {
537                XlogError::Kernel(format!("sync after compile_gpu_d4_gated failed: {}", e))
538            })?;
539        }
540    }
541    if circuit_base.num_nodes() == 0 || circuit_base.num_levels() == 0 {
542        // Defensive: the GPU-native Decision-DNNF compiler returned an empty circuit
543        // (the primary GPU cache hit is handled by the compile_needed == 0 early return
544        // above; this catches degenerate CNFs).
545        ledger.record_materialization()?;
546        let out_profile = if profiling { Some(profile) } else { None };
547        return Ok((handle, out_profile));
548    }
549
550    // --- Smoothing stage ---
551    let t_smooth = if profiling {
552        Some(Instant::now())
553    } else {
554        None
555    };
556    let circuit_eval = if random_vars.is_empty() {
557        circuit_base
558    } else {
559        #[cfg(debug_assertions)]
560        eprintln!("[xlog-prob] compile_gpu_d4_and_verify_cached: smooth_random_vars_device");
561        let smoothed = circuit_base.smooth_random_vars_device(
562            provider,
563            random_vars.list(),
564            random_vars.count(),
565            config.smooth_node_cap,
566            config.smooth_edge_cap,
567        )?;
568        #[cfg(debug_assertions)]
569        {
570            if !profiling {
571                provider.device().synchronize().map_err(|e| {
572                    XlogError::Kernel(format!(
573                        "sync after smooth_random_vars_device failed: {}",
574                        e
575                    ))
576                })?;
577            }
578        }
579        smoothed
580    };
581    if let Some(t0) = t_smooth {
582        provider
583            .device()
584            .synchronize()
585            .map_err(|e| XlogError::Kernel(format!("sync after smooth: {}", e)))?;
586        profile.smooth_sec = t0.elapsed().as_secs_f64();
587    }
588
589    // --- Verify final-circuit equivalence stage ---
590    //
591    // Certification is deliberately after every topology-changing pass. The
592    // exact `circuit_eval` value verified here is the value cached below and
593    // consumed by weighted model counting and gradients.
594    //
595    // `encode_cnf_gpu` sets `decision_var_limit` to the end of the leaf+choice var range. For
596    // deterministic programs with no probabilistic vars, this range is empty (limit=0). In that
597    // case, the verifier must still be able to branch, so fall back to `cnf.num_vars` (all CNF
598    // vars are semantically meaningful when there is no probabilistic decision set).
599    let verifier_decision_var_limit = if random_vars.is_empty() {
600        &cnf.num_vars
601    } else {
602        decision_var_limit
603    };
604    let cdcl = cdcl_config_from_compile(config)?;
605    #[cfg(debug_assertions)]
606    eprintln!("[xlog-prob] compile_gpu_d4_and_verify_cached: validate_equivalence_gpu_gated");
607    let t_verify = if profiling {
608        Some(Instant::now())
609    } else {
610        None
611    };
612    let certified_circuit = certify_final_circuit(
613        cnf,
614        verifier_decision_var_limit,
615        circuit_eval,
616        provider,
617        GpuEquivalenceConfig {
618            cdcl,
619            reuse_workspace: config.incremental_verify,
620        },
621        handle.compile_needed_device(),
622    )?;
623    if let Some(t0) = t_verify {
624        provider
625            .device()
626            .synchronize()
627            .map_err(|e| XlogError::Kernel(format!("sync after verify: {}", e)))?;
628        profile.verify_sec = t0.elapsed().as_secs_f64();
629    }
630    #[cfg(debug_assertions)]
631    {
632        if !profiling {
633            provider.device().synchronize().map_err(|e| {
634                XlogError::Kernel(format!(
635                    "sync after validate_equivalence_gpu_gated failed: {}",
636                    e
637                ))
638            })?;
639        }
640    }
641
642    // --- Cache store stage ---
643    #[cfg(debug_assertions)]
644    eprintln!("[xlog-prob] compile_gpu_d4_and_verify_cached: store_from_xgcf");
645    let t_store = if profiling {
646        Some(Instant::now())
647    } else {
648        None
649    };
650    cache.store_from_xgcf(&mut handle, certified_circuit.as_xgcf())?;
651    if let Some(t0) = t_store {
652        provider
653            .device()
654            .synchronize()
655            .map_err(|e| XlogError::Kernel(format!("sync after cache store: {}", e)))?;
656        profile.cache_store_sec = t0.elapsed().as_secs_f64();
657    }
658    #[cfg(debug_assertions)]
659    {
660        if !profiling {
661            provider.device().synchronize().map_err(|e| {
662                XlogError::Kernel(format!("sync after store_from_xgcf failed: {}", e))
663            })?;
664        }
665    }
666
667    // --- Free-var mask stage ---
668    #[cfg(debug_assertions)]
669    eprintln!("[xlog-prob] compile_gpu_d4_and_verify_cached: compute_free_var_mask_gpu_gated");
670    let t_fvm = if profiling {
671        Some(Instant::now())
672    } else {
673        None
674    };
675    let free_var_mask = gpu_d4::compute_free_var_mask_gpu_gated(
676        cnf,
677        certified_circuit.as_xgcf(),
678        provider,
679        handle.compile_needed_device(),
680    )?;
681    #[cfg(debug_assertions)]
682    {
683        if !profiling {
684            provider.device().synchronize().map_err(|e| {
685                XlogError::Kernel(format!(
686                    "sync after compute_free_var_mask_gpu_gated failed: {}",
687                    e
688                ))
689            })?;
690        }
691    }
692    // Only enable free-var correction if there are actual free variables.
693    // When the mask is all-zero (common for smoothed d-DNNF circuits),
694    // skipping this keeps has_free_var_mask[slot]=false, which avoids unnecessary
695    // free-var correction kernel launches on every subsequent eval.
696    let mask_host: Vec<u8> = provider
697        .device()
698        .inner()
699        .dtoh_sync_copy(&free_var_mask)
700        .map_err(|e| XlogError::Kernel(format!("Failed to read free_var_mask: {}", e)))?;
701    let has_free_vars = mask_host.iter().any(|&b| b != 0);
702    #[cfg(debug_assertions)]
703    eprintln!(
704        "[xlog-prob] free_var_mask: {} free vars, batched eval {}",
705        if has_free_vars { "has" } else { "no" },
706        if has_free_vars { "DISABLED" } else { "ENABLED" },
707    );
708    if has_free_vars {
709        cache.store_free_var_mask(&handle, &free_var_mask)?;
710        #[cfg(debug_assertions)]
711        {
712            if !profiling {
713                provider.device().synchronize().map_err(|e| {
714                    XlogError::Kernel(format!("sync after store_free_var_mask failed: {}", e))
715                })?;
716            }
717        }
718    }
719    if let Some(t0) = t_fvm {
720        provider
721            .device()
722            .synchronize()
723            .map_err(|e| XlogError::Kernel(format!("sync after free_var_mask: {}", e)))?;
724        profile.free_var_mask_sec = t0.elapsed().as_secs_f64();
725    }
726
727    // --- Disk cache write (opportunistic) ---
728    //
729    // After a successful compilation, write the artifact to disk for next warm start.
730    // Errors are silently ignored — the disk cache is best-effort.
731    if let Some(ref disk_key) = cache_key {
732        if let Ok(artifact) = cache.build_artifact_from_device(&handle, provider) {
733            let _ = disk_cache::write_artifact(disk_key, &artifact);
734            #[cfg(debug_assertions)]
735            eprintln!("[xlog-prob] compile_gpu_d4_and_verify_cached: wrote disk cache artifact");
736        }
737    }
738
739    ledger.record_materialization()?;
740    let out_profile = if profiling { Some(profile) } else { None };
741    Ok((handle, out_profile))
742}

744fn d4_config_for_smoothing(
745    config: &GpuCompileConfig,
746    random_var_count: u32,
747) -> Result<GpuCompileConfig> {
748    if random_var_count == 0 {
749        return Ok(*config);
750    }
751    let headroom = 2u32
752        .checked_add(random_var_count)
753        .ok_or_else(|| XlogError::Compilation("smooth headroom overflow".to_string()))?;
754    if config.smooth_node_cap <= headroom {
755        return Err(XlogError::Compilation(format!(
756            "GpuCompileConfig smooth_node_cap {} too small for smoothing headroom {}",
757            config.smooth_node_cap, headroom
758        )));
759    }
760    let base_cap = config
761        .smooth_node_cap
762        .checked_sub(headroom)
763        .ok_or_else(|| XlogError::Compilation("smooth node cap underflow".to_string()))?;
764    if base_cap < 3 {
765        return Err(XlogError::Compilation(
766            "GpuCompileConfig smooth_node_cap leaves <3 base nodes".to_string(),
767        ));
768    }
769    let mut out = *config;
770    out.smooth_node_cap = base_cap;
771    Ok(out)
772}

774fn cdcl_config_from_compile(config: &GpuCompileConfig) -> Result<GpuCdclConfig> {
775    if config.cdcl_restart_interval == 0 {
776        return Err(XlogError::Compilation(
777            "cdcl_restart_interval must be > 0".to_string(),
778        ));
779    }
780    if config.cdcl_learned_bytes == 0 {
781        return Err(XlogError::Compilation(
782            "cdcl_learned_bytes must be > 0".to_string(),
783        ));
784    }
785
786    // Deterministic sizing: assume average learned clause length = 4.
787    const AVG_LEN: u64 = 4;
788    const META_BYTES_PER_CLAUSE: u64 = 24; // offsets + lbd + activity + flags + proof offsets (rounded up)
789    const PROOF_BYTES_PER_CLAUSE: u64 = 8 + (8 * AVG_LEN); // (conflict, steps) + 2*u32 per lit
790    const LIT_BYTES_PER_CLAUSE: u64 = 4 * AVG_LEN;
791
792    let bytes_per_clause = META_BYTES_PER_CLAUSE
793        .checked_add(PROOF_BYTES_PER_CLAUSE)
794        .and_then(|v| v.checked_add(LIT_BYTES_PER_CLAUSE))
795        .ok_or_else(|| XlogError::Compilation("cdcl bytes per clause overflow".to_string()))?;
796
797    let max_clauses = config
798        .cdcl_learned_bytes
799        .checked_div(bytes_per_clause)
800        .ok_or_else(|| XlogError::Compilation("cdcl_learned_bytes div overflow".to_string()))?;
801    if max_clauses == 0 {
802        return Err(XlogError::Compilation(
803            "cdcl_learned_bytes too small for learned clause arena".to_string(),
804        ));
805    }
806
807    let max_lits = max_clauses
808        .checked_mul(AVG_LEN)
809        .ok_or_else(|| XlogError::Compilation("max_learned_lits overflow".to_string()))?;
810    let max_proof_u32 = max_clauses
811        .checked_mul(2 + 2 * AVG_LEN)
812        .ok_or_else(|| XlogError::Compilation("max_proof_u32 overflow".to_string()))?;
813
814    let max_learned_clauses = u32::try_from(max_clauses)
815        .map_err(|_| XlogError::Compilation("max_learned_clauses exceeds u32::MAX".to_string()))?;
816    let max_learned_lits = u32::try_from(max_lits)
817        .map_err(|_| XlogError::Compilation("max_learned_lits exceeds u32::MAX".to_string()))?;
818    let max_proof_u32 = u32::try_from(max_proof_u32)
819        .map_err(|_| XlogError::Compilation("max_proof_u32 exceeds u32::MAX".to_string()))?;
820
821    let reduce_interval = config
822        .cdcl_restart_interval
823        .checked_mul(20)
824        .ok_or_else(|| XlogError::Compilation("cdcl reduce_interval overflow".to_string()))?;
825
826    let mut gpu_cdcl = GpuCdclConfig::default();
827    gpu_cdcl.max_learned_clauses = max_learned_clauses;
828    gpu_cdcl.max_learned_lits = max_learned_lits;
829    gpu_cdcl.max_proof_u32 = max_proof_u32;
830    gpu_cdcl.restart_base = config.cdcl_restart_interval;
831    gpu_cdcl.reduce_interval = reduce_interval;
832    // Fail-closed conflict budget. Treewidth-hard equivalence verifies can
833    // spin past the GPU watchdog and crash with a context-poisoning launch
834    // failure; a finite budget makes them decline (VerifyBudgetExceeded) before
835    // that. Default 0 = unlimited (no behavior change). A recommended value is a
836    // calibration follow-up (between a completing verify's conflict count and
837    // the watchdog boundary); operators opt in via XLOG_D4_VERIFY_MAX_CONFLICTS.
838    gpu_cdcl.max_conflicts = verify_max_conflicts();
839    Ok(gpu_cdcl)
840}
841
842/// Per-verify conflict budget, read once from
843/// `XLOG_D4_VERIFY_MAX_CONFLICTS`. Default 0 = unlimited.
844fn verify_max_conflicts() -> u32 {
845    static BUDGET: std::sync::OnceLock<u32> = std::sync::OnceLock::new();
846    *BUDGET.get_or_init(|| {
847        std::env::var("XLOG_D4_VERIFY_MAX_CONFLICTS")
848            .ok()
849            .and_then(|v| v.trim().parse::<u32>().ok())
850            .unwrap_or(0)
851    })
852}

858/// FNV-1a 64-bit hash — deterministic across processes and Rust versions.
859/// Matches the FNV-1a algorithm used in the GPU hash kernel (kernels/cache.cu).
860fn fnv1a_u64(bytes: &[u8]) -> u64 {
861    const FNV_OFFSET: u64 = 0xcbf29ce484222325;
862    const FNV_PRIME: u64 = 0x100000001b3;
863    let mut h = FNV_OFFSET;
864    for &b in bytes {
865        h ^= b as u64;
866        h = h.wrapping_mul(FNV_PRIME);
867    }
868    h
869}

871fn combine_disk_cnf_hash(canonical_hash: u64, gpu_cnf_hash: u64) -> u64 {
872    let mut buf = Vec::with_capacity(16);
873    buf.extend_from_slice(&canonical_hash.to_le_bytes());
874    buf.extend_from_slice(&gpu_cnf_hash.to_le_bytes());
875    fnv1a_u64(&buf)
876}

878fn read_gpu_cnf_hash(
879    hash: &TrackedCudaSlice<u64>,
880    provider: &Arc<CudaKernelProvider>,
881) -> Result<u64> {
882    let host: Vec<u64> = provider
883        .device()
884        .inner()
885        .dtoh_sync_copy(hash)
886        .map_err(|e| XlogError::Kernel(format!("dtoh GPU CNF hash for disk cache key: {}", e)))?;
887    host.first()
888        .copied()
889        .ok_or_else(|| XlogError::Kernel("empty GPU CNF hash for disk cache key".to_string()))
890}

892/// Hash the compile config fields that affect circuit topology output.
893fn hash_compile_config(config: &GpuCompileConfig) -> u64 {
894    let mut buf = Vec::new();
895    buf.extend_from_slice(&config.frontier_depth.to_le_bytes());
896    buf.extend_from_slice(&config.max_frontier_items.to_le_bytes());
897    buf.extend_from_slice(&config.max_depth.to_le_bytes());
898    buf.extend_from_slice(&config.smooth_node_cap.to_le_bytes());
899    buf.extend_from_slice(&config.smooth_edge_cap.to_le_bytes());
900    // CDCL verifier params do not affect the compiled circuit topology,
901    // but we include them for safety so a verifier config change invalidates the cache.
902    buf.extend_from_slice(&config.cdcl_restart_interval.to_le_bytes());
903    buf.extend_from_slice(&config.cdcl_learned_bytes.to_le_bytes());
904    fnv1a_u64(&buf)
905}

907/// Hash the random variable list (D→H copy + hash).
908fn hash_random_vars(
909    random_vars: &DeviceRandomVarList,
910    provider: &Arc<CudaKernelProvider>,
911) -> Result<u64> {
912    let count = random_vars.count();
913    let mut buf = Vec::new();
914    buf.extend_from_slice(&count.to_le_bytes());
915    if count > 0 {
916        let host: Vec<u32> = provider
917            .device()
918            .inner()
919            .dtoh_sync_copy(random_vars.list())
920            .map_err(|e| {
921                XlogError::Kernel(format!("dtoh random_vars for disk cache hash: {}", e))
922            })?;
923        // Hash only the valid elements (count may be less than the allocation).
924        for &v in &host[..count as usize] {
925            buf.extend_from_slice(&v.to_le_bytes());
926        }
927    }
928    Ok(fnv1a_u64(&buf))
929}

931/// Query the device compute capability and encode as `major * 10 + minor` (e.g. 89 for sm_89).
932fn detect_compute_capability(provider: &Arc<CudaKernelProvider>) -> Result<u32> {
933    use cudarc::driver::sys::CUdevice_attribute;
934
935    let device = provider.device().inner();
936    let major = device
937        .attribute(CUdevice_attribute::CU_DEVICE_ATTRIBUTE_COMPUTE_CAPABILITY_MAJOR)
938        .map_err(|e| {
939            XlogError::Kernel(format!("Failed to query compute capability major: {}", e))
940        })?;
941    let minor = device
942        .attribute(CUdevice_attribute::CU_DEVICE_ATTRIBUTE_COMPUTE_CAPABILITY_MINOR)
943        .map_err(|e| {
944            XlogError::Kernel(format!("Failed to query compute capability minor: {}", e))
945        })?;
946    let major_u32: u32 = major.try_into().map_err(|_| {
947        XlogError::Kernel(format!(
948            "compute capability major {} cannot be converted to u32",
949            major
950        ))
951    })?;
952    let minor_u32: u32 = minor.try_into().map_err(|_| {
953        XlogError::Kernel(format!(
954            "compute capability minor {} cannot be converted to u32",
955            minor
956        ))
957    })?;
958    Ok(major_u32 * 10 + minor_u32)
