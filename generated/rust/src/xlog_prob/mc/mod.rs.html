1//! Approximate probabilistic inference via Monte Carlo sampling.
2//!
3//! This engine samples probabilistic facts / annotated disjunction decisions on the GPU and
4//! evaluates the deterministic core in each sampled world.
5//!
6//! For programs with non-monotone recursion (cycles through `not` and/or aggregates),
7//! Monte Carlo evaluation requires the user to opt into the approximate probabilistic
8//! engine. The deterministic evaluation uses a bounded, cycle-aware semantics:
9//!
10//! - If an SCC reaches a fixpoint under synchronous iteration, that fixpoint is used.
11//! - If the SCC enters a cycle, the interpretation is the intersection of all states in the cycle
12//!   (skeptical, invariant tuples only). This avoids parity/oscillation dependence on iteration
13//!   count while remaining fully deterministic and explicit.

51/// Sampling method for Monte Carlo inference.
52#[derive(Debug, Clone, Copy, PartialEq, Eq)]
53pub enum McSamplingMethod {
54    /// Sample from prior, discard worlds where evidence is not satisfied.
55    Rejection,
56    /// Force evidence variables in the sampler; every sample counts.
57    EvidenceClamping,
58}
59
60impl McSamplingMethod {
61    /// Returns the stable machine-readable sampling-method name.
62    pub fn as_str(self) -> &'static str {
63        match self {
64            McSamplingMethod::Rejection => "rejection",
65            McSamplingMethod::EvidenceClamping => "evidence_clamping",
66        }
67    }
68}
69
70impl From<ProbMethod> for McSamplingMethod {
71    fn from(value: ProbMethod) -> Self {
72        match value {
73            ProbMethod::Rejection => Self::Rejection,
74            ProbMethod::EvidenceClamping => Self::EvidenceClamping,
75        }
76    }
77}

79/// Strategy for counting evidence-satisfied samples in the MC loop.
80///
81/// In `QueriesOnly` mode (used with evidence clamping), evidence is
82/// guaranteed to hold in every sample, so we skip the truth-kernel's
83/// evidence check and evidence-side buffer allocations.
84#[derive(Debug, Clone, Copy, PartialEq, Eq)]
85pub enum McCountStrategy {
86    /// Full path: check both queries and evidence each sample.
87    QueriesAndEvidence,
88    /// Clamped path: evidence is always satisfied; only accumulate query flags.
89    QueriesOnly,
90}
91
92impl McCountStrategy {
93    /// Derive the count strategy from the chosen sampling method.
94    pub fn from_method(method: McSamplingMethod) -> Self {
95        match method {
96            McSamplingMethod::Rejection => Self::QueriesAndEvidence,
97            McSamplingMethod::EvidenceClamping => Self::QueriesOnly,
98        }
99    }
100}

102/// Breakdown of time spent in each phase of MC evaluation.
103/// Gate with `XLOG_MC_PROFILE=1` to print at the end of evaluation.
104#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
105pub struct McTimingBreakdown {
106    /// Microseconds spent drawing Bernoulli samples.
107    pub sampler_us: u64,
108    /// Microseconds spent resetting per-sample state.
109    pub sample_reset_us: u64,
110    /// Microseconds spent materializing sampled extensional facts.
111    pub sample_build_us: u64,
112    /// Microseconds spent evaluating deterministic rules.
113    pub eval_us: u64,
114    /// Microseconds spent accumulating evidence and query counts.
115    pub count_us: u64,
116}
117
118impl McTimingBreakdown {
119    /// Returns the saturating sum of all measured phases.
120    pub fn total_us(&self) -> u64 {
121        self.sampler_us
122            .saturating_add(self.sample_reset_us)
123            .saturating_add(self.sample_build_us)
124            .saturating_add(self.eval_us)
125            .saturating_add(self.count_us)
126    }
127}

134/// Configuration for Monte Carlo probabilistic inference.
135///
136/// Use [`McEvalConfig::default()`] as a starting point and then update the
137/// individual fields you need.
138pub struct McEvalConfig {
139    /// Number of Monte Carlo samples.
140    pub samples: usize,
141    /// RNG seed (deterministic).
142    pub seed: u64,
143    /// Two-sided confidence level in (0,1) (e.g., 0.95).
144    pub confidence: f64,
145    /// Maximum SCC iteration steps for non-monotone cycle detection.
146    pub max_nonmonotone_iterations: usize,
147    /// Sampling method override. `None` = auto-select (EvidenceClamping when forceable, Rejection otherwise).
148    pub sampling_method: Option<McSamplingMethod>,
149    /// Allow the host CPU oracle ([`McProgram::evaluate_cpu`]) when the
150    /// resident GPU engine rejects the program. Default `false`: a rejected
151    /// program fails closed with the typed rejection instead of silently
152    /// running on the CPU. When set, the oracle result is labeled
153    /// [`McEngine::CpuOracle`] and must never serve as GPU-native evidence.
154    pub allow_cpu_oracle_fallback: bool,
155}

189impl McEvalConfig {
190    /// Resolves Monte Carlo settings from program directives and validates them.
191    pub fn from_directives(directives: &Directives) -> Result<Self> {
192        let mut cfg = Self::default();
193        if let Some(samples) = directives.prob_samples {
194            cfg.samples = samples;
195        }
196        if let Some(seed) = directives.prob_seed {
197            cfg.seed = seed;
198        }
199        if let Some(confidence) = directives.prob_confidence {
200            cfg.confidence = confidence;
201        }
202        if let Some(iterations) = directives.prob_max_nonmonotone_iterations {
203            cfg.max_nonmonotone_iterations = iterations;
204        }
205        cfg.sampling_method = directives.prob_method.map(McSamplingMethod::from);
206        cfg.validate()?;
207        Ok(cfg)
208    }
209
210    /// Apply only explicitly supplied caller values, then validate the resolved config.
211    pub fn apply_overrides(&mut self, overrides: McEvalOverrides) -> Result<()> {
212        if let Some(samples) = overrides.samples {
213            self.samples = samples;
214        }
215        if let Some(seed) = overrides.seed {
216            self.seed = seed;
217        }
218        if let Some(confidence) = overrides.confidence {
219            self.confidence = confidence;
220        }
221        if let Some(iterations) = overrides.max_nonmonotone_iterations {
222            self.max_nonmonotone_iterations = iterations;
223        }
224        if let Some(method) = overrides.sampling_method {
225            self.sampling_method = Some(method);
226        }
227        if let Some(allow) = overrides.allow_cpu_oracle_fallback {
228            self.allow_cpu_oracle_fallback = allow;
229        }
230        self.validate()
231    }
232
233    /// Validates numeric ranges and required positive limits.
234    pub fn validate(&self) -> Result<()> {
235        if self.samples == 0 {
236            return Err(XlogError::Compilation(
237                "MC inference requires samples > 0".to_string(),
238            ));
239        }
240        if !(0.0 < self.confidence && self.confidence < 1.0) || self.confidence.is_nan() {
241            return Err(XlogError::Compilation(format!(
242                "MC inference requires 0 < confidence < 1, got {}",
243                self.confidence
244            )));
245        }
246        if self.max_nonmonotone_iterations == 0 {
247            return Err(XlogError::Compilation(
248                "MC inference requires max_nonmonotone_iterations > 0".to_string(),
249            ));
250        }
251        Ok(())
252    }
253}

320/// **Legacy back-compat surface** — tracked (data-plane) host<->device transfer
321/// deltas measured around the MC measured region.
322///
323/// This struct dates from the predecessor `a894aab4` engine, which removed only
324/// *tracked* data-plane transfers from a still-host-orchestrated loop. The
325/// current resident megakernel engine has a *stronger* property — **no host
326/// interaction at all** in the measured region — so the authoritative contract
327/// now lives in [`McNoHostStats`] (`McResidentResult::no_host`), which also
328/// counts untracked metadata reads, host fixpoint iterations, and in-region
329/// device allocations. `McDeviceResult` retains this field for API
330/// back-compatibility; for the resident engine its tracked-call fields are zero.

334    pub htod_calls: u64,
335    /// Tracked device-to-host calls in the measured region.
336    pub dtoh_calls: u64,
337    /// Tracked host-to-device bytes in the measured region.
338    pub htod_bytes: u64,
339    /// Tracked device-to-host bytes in the measured region.
340    pub dtoh_bytes: u64,
341}

343impl McHotLoopTransfers {

485    /// Host-facing MC evaluation: runs the resident megakernel engine
486    /// ([`Self::evaluate_gpu_device_with_provider`]) and then **materializes the
487    /// result on the host** by downloading the final query/evidence counts
488    /// *after* the measured region. The download is a host-result
489    /// materialization, not part of the measured region — the no-host property
490    /// belongs to the resident engine, not to this convenience wrapper. Use
491    /// [`Self::evaluate_gpu_device`] when you want device-resident counts with no
492    /// host download at all.
493    ///
494    /// **Fail-closed contract:** if the resident engine rejects the program
495    /// (negation, aggregates, unbounded terms, ...), this returns the typed
496    /// rejection error. The CPU oracle is reachable only via the explicit
497    /// [`McEvalConfig::allow_cpu_oracle_fallback`] opt-in and its result is
498    /// labeled [`McEngine::CpuOracle`].
499    #[cfg(feature = "host-io")]
500    pub fn evaluate(&self, cfg: McEvalConfig) -> Result<McResult> {
501        let provider = Arc::new(self.provider()?);
502        let cfg_clone = cfg.clone();
503        let device_result =
504            match self.evaluate_gpu_device_with_provider(cfg_clone, provider.clone()) {
505                Ok(result) => result,
506                Err(XlogError::Compilation(message))
507                    if message.starts_with("resident MC engine rejected program") =>
508                {
509                    // Fail closed by default: the CPU oracle is an explicit,
510                    // labeled opt-in (`allow_cpu_oracle_fallback`), never a
511                    // silent substitute for the resident GPU engine.
512                    if cfg.allow_cpu_oracle_fallback {
513                        return self.evaluate_cpu(cfg);
514                    }
515                    return Err(XlogError::Compilation(format!(
516                        "{message}; MC fail-closed: set \
517                         McEvalConfig::allow_cpu_oracle_fallback to explicitly \
518                         run the labeled CPU oracle instead"
519                    )));
520                }
521                Err(err) => return Err(err),
522            };
523
524        let mut host_counts = vec![0u32; device_result.query_counts.len()];
525        if !host_counts.is_empty() {
526            provider
527                .device()
528                .inner()
529                .dtoh_sync_copy_into(&device_result.query_counts, &mut host_counts)
530                .map_err(|e| {
531                    XlogError::Kernel(format!("Failed to download MC query counts: {}", e))
532                })?;
533        }
534
535        let mut host_evidence = [0u32];
536        provider
537            .device()
538            .inner()
539            .dtoh_sync_copy_into(&device_result.evidence_count, &mut host_evidence)
540            .map_err(|e| {
541                XlogError::Kernel(format!("Failed to download MC evidence count: {}", e))
542            })?;
543
544        let evidence_samples = if self.evidence.is_empty() {
545            cfg.samples
546        } else {
547            host_evidence[0] as usize
548        };
549
550        if device_result.sampling_method != McSamplingMethod::EvidenceClamping
551            && !self.evidence.is_empty()
552            && evidence_samples == 0
553        {
554            return Err(XlogError::Execution(format!(
555                "MC inference error: evidence was never satisfied across {} samples (seed={})",
556                cfg.samples, cfg.seed
557            )));
558        }
559
560        let z = results::normal_quantile(0.5 + cfg.confidence / 2.0);
561        let mut query_estimates: Vec<McQueryEstimate> = Vec::with_capacity(self.queries.len());
562        for (i, atom) in self.query_presentations.iter().enumerate() {
563            let k = host_counts.get(i).copied().unwrap_or(0) as usize;
564            let (p, stderr, ci_low, ci_high) = results::binomial_estimate(k, evidence_samples, z);
565            let log_prob = if p == 0.0 { f64::NEG_INFINITY } else { p.ln() };
566
567            query_estimates.push(McQueryEstimate {
568                atom: atom.clone(),
569                prob: p,
570                log_prob,
571                stderr,
572                ci_low,
573                ci_high,
574            });
575        }
576
577        Ok(McResult {
578            total_samples: cfg.samples,
579            evidence_samples,
580            seed: cfg.seed,
581            confidence: cfg.confidence,
582            query_estimates,
583            nonmonotone_sccs: device_result.nonmonotone_sccs,
584            nonmonotone_cycles: device_result.nonmonotone_cycles,
585            engine: McEngine::GpuResident,
586            nonmonotone_iteration_limit_hits: device_result.nonmonotone_iteration_limit_hits,
587            sampling_method: device_result.sampling_method,
588        })
589    }

591    /// CPU **oracle / debug** MC path. Downloads the full sampled-bit matrix to
592    /// the host and evaluates every sampled world on a host relation store.
593    ///
594    /// This is intentionally *not* GPU-native: it performs a large DtoH of the
595    /// sample matrix and runs the deterministic core on the CPU. It exists solely
596    /// as a deterministic, seed-matched oracle for validating the GPU-native
597    /// device counts (the GPU sampler is shared, so for the same program/seed the
598    /// two paths see identical samples). It must **never** be used as zero-host /
599    /// GPU-native release evidence, and the acceptance matrix excludes it and the
600    /// tests that call it (`tests/gpu_mc_vs_cpu.rs`, `tests/mc.rs`).
601    #[cfg(feature = "host-io")]
602    pub fn evaluate_cpu(&self, cfg: McEvalConfig) -> Result<McResult> {
603        if cfg.samples == 0 {
604            return Err(XlogError::Execution(
605                "MC inference requires samples > 0".to_string(),
606            ));
607        }
608        if !(0.0 < cfg.confidence && cfg.confidence < 1.0) || cfg.confidence.is_nan() {
609            return Err(XlogError::Execution(format!(
610                "MC inference requires 0 < confidence < 1, got {}",
611                cfg.confidence
612            )));
613        }
614        if cfg.max_nonmonotone_iterations == 0 {
615            return Err(XlogError::Execution(
616                "MC inference requires max_nonmonotone_iterations > 0".to_string(),
617            ));
618        }
619
620        let (method, forcing) = self.resolve_sampling_method(cfg.sampling_method)?;
621        let is_clamped = method == McSamplingMethod::EvidenceClamping;
622
623        let mut n_evidence: usize = 0;
624        let mut n_query_true: Vec<usize> = vec![0; self.queries.len()];
625        let mut stats = EvalStats::default();
626
627        let num_vars = self.bernoulli_probs.len();
628        let samples_matrix: Vec<u8> = if num_vars == 0 {
629            Vec::new()
630        } else {
631            let total = num_vars
632                .checked_mul(cfg.samples)
633                .ok_or_else(|| XlogError::Execution("MC sample matrix overflow".to_string()))?;
634            let provider = Arc::new(self.provider()?);
635
636            // Allocate force arrays: upload actual forcing data in clamped mode, zero-fill otherwise
637            let mut d_force_mask = provider.memory().alloc::<u8>(num_vars.max(1))?;
638            let mut d_forced_value = provider.memory().alloc::<u8>(num_vars.max(1))?;
639            if is_clamped {
640                provider
641                    .htod_sync_copy_into_tracked(&forcing.force_mask, &mut d_force_mask)
642                    .map_err(|e| {
643                        XlogError::Kernel(format!("Failed to upload force_mask: {}", e))
644                    })?;
645                provider
646                    .htod_sync_copy_into_tracked(&forcing.forced_value, &mut d_forced_value)
647                    .map_err(|e| {
648                        XlogError::Kernel(format!("Failed to upload forced_value: {}", e))
649                    })?;
650            } else {
651                provider
652                    .device()
653                    .inner()
654                    .memset_zeros(&mut d_force_mask)
655                    .map_err(|e| XlogError::Kernel(format!("Failed to zero force_mask: {}", e)))?;
656                provider
657                    .device()
658                    .inner()
659                    .memset_zeros(&mut d_forced_value)
660                    .map_err(|e| {
661                        XlogError::Kernel(format!("Failed to zero forced_value: {}", e))
662                    })?;
663            }
664
665            let samples_device = provider.sample_bernoulli_matrix_device(
666                &self.bernoulli_probs,
667                cfg.samples,
668                cfg.seed,
669                &d_force_mask.slice(..),
670                &d_forced_value.slice(..),
671            )?;
672            let mut host = vec![0u8; total];
673            if !host.is_empty() {
674                provider
675                    .device()
676                    .inner()
677                    .dtoh_sync_copy_into(&samples_device, &mut host)
678                    .map_err(|e| {
679                        XlogError::Kernel(format!("Failed to download MC samples: {}", e))
680                    })?;
681            }
682            host
683        };
684
685        for sample_idx in 0..cfg.samples {
686            let sample_bits = if num_vars == 0 {
687                &[][..]
688            } else {
689                let start = sample_idx * num_vars;
690                let end = start + num_vars;
691                &samples_matrix[start..end]
692            };
693
694            let mut store = self.base_store.clone();
695            self.apply_sample_facts(&mut store, sample_bits)?;
696
697            let sample_stats = results::evaluate_program_inplace(
698                &self.scc_plans,
699                &mut store,
700                cfg.max_nonmonotone_iterations,
701                &self.arithmetic_schemas,
702            )?;
703            stats.nonmonotone_sccs += sample_stats.nonmonotone_sccs;
704            stats.nonmonotone_cycles += sample_stats.nonmonotone_cycles;
705            stats.nonmonotone_iteration_limit_hits += sample_stats.nonmonotone_iteration_limit_hits;
706
707            // In clamped mode, skip evidence check — all samples count
708            if !is_clamped && !results::evidence_satisfied(&store, &self.evidence) {
709                continue;
710            }
711
712            n_evidence += 1;
713            for (i, q) in self.queries.iter().enumerate() {
714                if results::atom_holds(&store, q) {
715                    n_query_true[i] += 1;
716                }
717            }
718        }
719
720        if !is_clamped && !self.evidence.is_empty() && n_evidence == 0 {
721            return Err(XlogError::Execution(format!(
722                "MC inference error: evidence was never satisfied across {} samples (seed={})",
723                cfg.samples, cfg.seed
724            )));
725        }
726
727        // If there is no evidence (or clamped mode), treat all samples as evidence-satisfying.
728        let denom = if self.evidence.is_empty() || is_clamped {
729            cfg.samples
730        } else {
731            n_evidence
732        };
733
734        let z = results::normal_quantile(0.5 + cfg.confidence / 2.0);
735
736        let mut query_estimates: Vec<McQueryEstimate> = Vec::with_capacity(self.queries.len());
737        for (i, atom) in self.query_presentations.iter().enumerate() {
738            let k = n_query_true[i];
739            let (p, stderr, ci_low, ci_high) = results::binomial_estimate(k, denom, z);
740            let log_prob = if p == 0.0 { f64::NEG_INFINITY } else { p.ln() };
741
742            query_estimates.push(McQueryEstimate {
743                atom: atom.clone(),
744                prob: p,
745                log_prob,
746                stderr,
747                ci_low,
748                ci_high,
749            });
750        }
751
752        Ok(McResult {
753            total_samples: cfg.samples,
754            evidence_samples: denom,
755            seed: cfg.seed,
756            confidence: cfg.confidence,
757            query_estimates,
758            nonmonotone_sccs: stats.nonmonotone_sccs,
759            nonmonotone_cycles: stats.nonmonotone_cycles,
760            nonmonotone_iteration_limit_hits: stats.nonmonotone_iteration_limit_hits,
761            sampling_method: method,
762            engine: McEngine::CpuOracle,
763        })
764    }

776    /// GPU-native device-resident MC evaluation via the resident megakernel
777    /// engine ([`resident`]). Returns [`McDeviceResult`] with counts left on the
778    /// device (no host download). The engine evaluates all worlds in a single
779    /// launch with **no host interaction in the measured region** (no host
780    /// sample loop, no per-sample/per-operator host launches or allocations, no
781    /// tracked transfers, and no untracked metadata reads); see
782    /// [`McResidentResult::no_host`] / [`McNoHostStats::is_no_host`] for the full
783    /// measured contract.
784    pub fn evaluate_gpu_device(&self, cfg: McEvalConfig) -> Result<McDeviceResult> {
785        let provider = Arc::new(self.provider()?);
786        self.evaluate_gpu_device_with_provider(cfg, provider)
787    }

789    /// GPU-native device-resident MC evaluation using a caller-supplied provider
790    /// (enables provider/buffer reuse across calls). Static setup (arena
791    /// allocation, plan upload, sampling) happens before the measured region;
792    /// the measured region is a single resident-engine launch with **zero host
793    /// interaction** — no host loop over samples or fixpoint iterations, no
794    /// per-sample/per-operator host launches or device allocations, no tracked
795    /// HtoD/DtoH transfers, and no untracked metadata reads. The full contract is
796    /// measured by [`McNoHostStats`] (`McResidentResult::no_host`) and gated by
797    /// `tests/mc_resident.rs`. Counts remain device-resident; the caller decides
798    /// whether/when to download them.
799    pub fn evaluate_gpu_device_with_provider(
800        &self,
801        cfg: McEvalConfig,
802        provider: Arc<CudaKernelProvider>,
803    ) -> Result<McDeviceResult> {
804        // The GPU-resident megakernel engine is the sole MC execution path: there
805        // is no host-orchestrated per-sample fallback. It evaluates ALL worlds in
806        // a single launch with zero host interaction in the measured region, then
807        // returns device-resident counts. Programs outside the supported bounded
808        // fragment fail closed (typed `ResidentRejection`).
809        let r = self.evaluate_resident_with_provider(cfg, provider)?;
810        let no_host = r.no_host;
811        Ok(McDeviceResult {
812            query_counts: r.query_counts,
813            evidence_count: r.evidence_count,
814            total_samples: r.total_samples,
815            seed: r.seed,
816            confidence: r.confidence,
817            // The resident engine accepts only bounded positive Datalog and
818            // device-evaluates its fixpoint; legacy non-monotone SCC bookkeeping
819            // is not part of this engine's reported state.
820            nonmonotone_sccs: 0,
821            nonmonotone_cycles: 0,
822            nonmonotone_iteration_limit_hits: 0,
823            sampling_method: r.sampling_method,
824            // Back-compat surface: `hot_loop_transfers` reports the resident
825            // engine's tracked transfers (zero). Richer no-host evidence (untracked
826            // reads, fixpoint/alloc counts) lives in `McResidentResult::no_host`.
827            hot_loop_transfers: McHotLoopTransfers {
828                htod_calls: no_host.tracked_htod_calls,
829                dtoh_calls: no_host.tracked_dtoh_calls,
830                htod_bytes: 0,
831                dtoh_bytes: 0,
832            },
833            no_host,
834        })
835    }

837    fn compile_program(program: &Program) -> Result<Self> {
838        #[cfg(feature = "host-io")]
839        let source_program = program;
840        let arithmetic_schemas = crate::provenance::arithmetic_schemas(program)?;
841        let canonical_program = canonicalize_probabilistic_program(program, &arithmetic_schemas)?;
842        let program = &canonical_program;
843
844        let mut queries: Vec<GroundAtom> = Vec::new();
845        for ProbQuery { atom } in &program.prob_queries {
846            queries.push(atom_key_from_ground_atom(atom)?);
847        }
848        #[cfg(feature = "host-io")]
849        let mut query_presentations: Vec<GroundAtom> = Vec::new();
850        #[cfg(feature = "host-io")]
851        {
852            for (ProbQuery { atom }, ProbQuery { atom: source_atom }) in program
853                .prob_queries
854                .iter()
855                .zip(&source_program.prob_queries)
856            {
857                query_presentations.push(presentation_atom_from_canonical(
858                    source_atom,
859                    atom,
860                    &arithmetic_schemas,
861                )?);
862            }
863        }
864
865        let mut evidence: Vec<(GroundAtom, bool)> = Vec::new();
866        let mut evidence_values: BTreeMap<GroundAtom, bool> = BTreeMap::new();
867        for Evidence { atom, value } in &program.evidence {
868            let atom = atom_key_from_ground_atom(atom)?;
869            match evidence_values.entry(atom.clone()) {
870                std::collections::btree_map::Entry::Occupied(previous)
871                    if previous.get() != value =>
872                {
873                    return Err(XlogError::Execution(format!(
874                        "Monte Carlo inference error: conflicting evidence for {}",
875                        atom.predicate
876                    )));
877                }
878                std::collections::btree_map::Entry::Occupied(_) => continue,
879                std::collections::btree_map::Entry::Vacant(entry) => {
880                    entry.insert(*value);
881                    evidence.push((atom, *value));
882                }
883            }
884        }
885
886        let mut prob_facts = program.prob_facts.clone();
887        buffers::extend_prob_facts_with_coin(program, &mut prob_facts)?;
888        let (bernoulli_probs, prob_facts, annotated_disjunctions) =
889            buffers::compile_sampling_plan(&prob_facts, &program.annotated_disjunctions)?;
890
891        #[cfg(feature = "host-io")]
892        let mut base_store: HashMap<String, Relation> = HashMap::new();
893
894        // Deterministic facts.
895        #[cfg(feature = "host-io")]
896        {
897            for fact in program.facts() {
898                let atom = atom_key_from_ground_atom(&fact.head)?;
899                base_store
900                    .entry(atom.predicate.clone())
901                    .or_default()
902                    .insert_tuple(atom.args);
903            }
904        }
905
906        // Ensure relations exist for all referenced predicates so evaluation treats missing as empty,
907        // but never errors due to an unknown predicate (CPU eval path only).
908        #[cfg(feature = "host-io")]
909        {
910            let mut referenced: HashSet<String> = HashSet::new();
911            for rule in &program.rules {
912                referenced.insert(rule.head.predicate.clone());
913                for lit in &rule.body {
914                    match lit {
915                        BodyLiteral::Positive(a) | BodyLiteral::Negated(a) => {
916                            referenced.insert(a.predicate.clone());
917                        }
918                        BodyLiteral::Epistemic(lit) => {
919                            referenced.insert(lit.atom.predicate.clone());
920                        }
921                        BodyLiteral::Comparison(_)
922                        | BodyLiteral::IsExpr(_)
923                        | BodyLiteral::Univ(_) => {}
924                    }
925                }
926            }
927            for pf in &program.prob_facts {
928                referenced.insert(pf.atom.predicate.clone());
929            }
930            for ad in &program.annotated_disjunctions {
931                for pf in &ad.choices {
932                    referenced.insert(pf.atom.predicate.clone());
933                }
934            }
935            for q in &queries {
936                referenced.insert(q.predicate.clone());
937            }
938            for (e, _) in &evidence {
939                referenced.insert(e.predicate.clone());
940            }
941            for pred in referenced {
942                base_store.entry(pred).or_default();
943            }
944        }
945
946        #[cfg(feature = "host-io")]
947        let scc_plans = results::build_scc_plans(program)?;
948        Ok(Self {
949            gpu_config: GpuConfig::default(),
950            program: program.clone(),
951            #[cfg(feature = "host-io")]
952            base_store,
953            #[cfg(feature = "host-io")]
954            scc_plans,
955            #[cfg(feature = "host-io")]
956            arithmetic_schemas,
957            queries,
958            #[cfg(feature = "host-io")]
959            query_presentations,
960            evidence,
961            bernoulli_probs,
962            prob_facts,
963            annotated_disjunctions,
964        })
965    }

967    fn provider(&self) -> Result<CudaKernelProvider> {
968        if self.gpu_config.memory_bytes == 0 {
969            return Err(XlogError::Kernel(
970                "GPU memory budget must be non-zero".to_string(),
971            ));
972        }
973
974        xlog_cuda::CudaProviderBuilder::new(
975            self.gpu_config.device_ordinal,
976            MemoryBudget::with_limit(self.gpu_config.memory_bytes),
977        )
978        .build()
979    }
