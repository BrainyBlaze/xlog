2use pyo3::types::{PyDict, PyList};

4pub(crate) fn pack_rule_provenance(
5    py: Python<'_>,
6    entries: &[xlog_logic::RuleProvenance],
7) -> PyResult<Py<PyAny>> {
8    let list = PyList::empty(py);
9    for entry in entries {
10        let dict = PyDict::new(py);
11        dict.set_item("rule_id", &entry.rule_id)?;
12        dict.set_item("head", &entry.head)?;
13        dict.set_item("source_kind", entry.source_kind.as_str())?;
14        match &entry.source_span {
15            Some(source_span) => dict.set_item("source_span", source_span)?,
16            None => dict.set_item("source_span", py.None())?,
17        }
18        match &entry.generation_trace_hash {
19            Some(hash) => dict.set_item("generation_trace_hash", hash)?,
20            None => dict.set_item("generation_trace_hash", py.None())?,
21        }
22        dict.set_item("support_relation_ids", &entry.support_relation_ids)?;
23        dict.set_item(
24            "counterexample_relation_ids",
25            &entry.counterexample_relation_ids,
26        )?;
27        list.append(dict)?;
28    }
29    Ok(list.into())
30}
