36/// A cached circuit for a specific query template.
37///
38/// The circuit structure is immutable - only weights change between queries.
39/// Weight slots map network outputs to circuit variables.
40pub(crate) struct CachedCircuit {
41    /// The compiled program containing the GPU circuit
42    pub(crate) program: ExactDdnnfProgram,
43
44    /// Device-resident mapping from neural output slots to CNF variable ids.
45    pub(crate) slots: GpuWeightSlots,
46
47    /// Ordered target domain for Targeted signatures. Empty for Boolean.
48    pub(crate) target_domain: Vec<String>,
49}

51#[derive(Debug, Clone)]
52pub(crate) enum InputSource {
53    QueryArg(usize),
54    ImplicitSlot(usize),
55    /// Stage-B real-domain grounding: read row `usize` of the join-domain tensor
56    /// source (`nsr_domain`, the per-event feature batch). Used by the per-event
57    /// expansions of a neural predicate joined on an existential variable.

66pub(crate) struct NeuralGroup {
67    pub(crate) info: NeuralPredicateInfo,
68    pub(crate) input_source: InputSource,
69    /// Stage-B real-domain grounding: when `Some(c)`, the template grounds this
70    /// group's neural atom at the real constant `c` (a domain event id or a

74    #[cfg(feature = "host-io")]
75    pub(crate) output_var: Option<String>,

78/// An ordinary-relation body atom in a trainable-rule query treated as a HARD
79/// join condition: it gates which query groundings can fire but contributes no
80/// probability mass and no gradient. The query probability is
81/// `(hard conditions satisfiable?) x (neural prob)`; gradients flow only
82/// through the neural predicates x sigma(w), never through these fact atoms.
83#[derive(Debug, Clone)]
84pub(crate) struct HardFilter {
85    /// Ordinary relation name (must hold over the program's facts).
86    pub(crate) relation: String,
87    /// For each relation argument position, the query HEAD position whose value
88    /// it must equal. Current scope: every relation argument is a head
89    /// variable; hard conditions that join on existential (non-head) variables
90    /// are a documented follow-up.
91    pub(crate) arg_head_positions: Vec<usize>,
92}
93
94/// Stage-B existential join: the plan for grounding a neural predicate over the
95/// REAL join domain inside the circuit (instead of stripping the join relation as
96/// a pre-filter). The neural groups are already expanded per domain constant; this
97/// records the ordinary relations whose ground facts must stay IN the circuit (so
98/// provenance OR-aggregates `OR_event(neural(event) ∧ join(event, head))` at each
99/// head binding) and the head-key domain that the query ranges over.
100#[derive(Debug, Clone)]
101pub(crate) struct JoinPlan {
102    /// Ordinary relations kept inside the circuit rule; their ground facts are
103    /// added to the template program (read from `self.ast`).
104    pub(crate) relations: Vec<String>,
105    /// The real head-key domain the query ranges over (e.g. edge ids), in the
106    /// same sorted order as the per-edge guard group expansion and the emitted
107    /// `prob_queries`. Serves as the `target_domain` for a join signature.
108    pub(crate) head_domain: Vec<String>,
109}

111#[derive(Debug, Clone)]
112pub(crate) enum QuerySignature {
113    Boolean {
114        groups: Vec<NeuralGroup>,
115        hard_filters: Vec<HardFilter>,
116    },
117    Targeted {
118        target_position: usize,
119        groups: Vec<NeuralGroup>,
120        hard_filters: Vec<HardFilter>,
121        /// `Some` for a Stage-B existential-join signature: the head target
122        /// position ranges over the real head-key domain (e.g. edges) and the
123        /// groups are real-domain-grounded. `None` for the ordinary targeted path.
124        join: Option<JoinPlan>,
125    },
126}

128impl QuerySignature {
129    pub(crate) fn groups(&self) -> &[NeuralGroup] {
130        match self {
131            QuerySignature::Boolean { groups, .. } | QuerySignature::Targeted { groups, .. } => {
132                groups
133            }
134        }
135    }
136
137    pub(crate) fn hard_filters(&self) -> &[HardFilter] {
138        match self {
139            QuerySignature::Boolean { hard_filters, .. }
140            | QuerySignature::Targeted { hard_filters, .. } => hard_filters,
141        }
142    }
143
144    /// The Stage-B join plan, if this is an existential-join signature.
145    pub(crate) fn join(&self) -> Option<&JoinPlan> {
146        match self {
147            QuerySignature::Targeted { join, .. } => join.as_ref(),
148            QuerySignature::Boolean { .. } => None,
149        }
150    }

172#[cfg(feature = "host-io")]
173pub(crate) fn atom_to_string(atom: &xlog_prob::provenance::GroundAtom) -> String {
174    use xlog_prob::provenance::Value;
175
176    if atom.args.is_empty() {
177        return format!("{}()", atom.predicate);
178    }
179
180    let mut s = String::new();
181    s.push_str(&atom.predicate);
182    s.push('(');
183    for (i, arg) in atom.args.iter().enumerate() {
184        if i != 0 {
185            s.push_str(", ");
186        }
187        match arg {
188            Value::I64(v) => s.push_str(&v.to_string()),
189            Value::F64(bits) => s.push_str(&f64::from_bits(*bits).to_string()),
190            Value::Symbol(sym) => {
191                s.push_str(&symbol::resolve_checked(*sym).unwrap_or_else(|| format!("sym#{}", sym)))
192            }
193            Value::String(v) => s.push_str(v),
194        }
195    }
196    s.push(')');
197    s
198}

200fn resolve_mc_eval_config(
201    directives: &xlog_logic::ast::Directives,
202    samples: Option<usize>,
203    seed: Option<u64>,
204    confidence: Option<f64>,
205    max_nonmonotone_iterations: Option<usize>,
206    sampling_method: Option<McSamplingMethod>,
207    allow_cpu_oracle_fallback: bool,
208) -> xlog_core::Result<McEvalConfig> {
209    let mut cfg = McEvalConfig::from_directives(directives)?;
210    cfg.apply_overrides(McEvalOverrides {
211        samples,
212        seed,
213        confidence,
214        max_nonmonotone_iterations,
215        sampling_method,
216        allow_cpu_oracle_fallback: Some(allow_cpu_oracle_fallback),
217    })?;
218    Ok(cfg)
219}

232                "Unknown sampling_method '{}'. Use 'rejection' or 'evidence_clamping'.",
233                other
234            ))),
235        }

238    /// Evaluate probability of a single query by compiling a temporary program.
239    pub(crate) fn evaluate_query_probability(&self, query: &str) -> PyResult<f64> {
240        let probs = self.evaluate_query_probabilities(&[query.to_string()])?;
241        probs
242            .into_iter()
243            .next()
244            .ok_or_else(|| PyRuntimeError::new_err("Query evaluation returned no results"))
245    }

247    /// Evaluate probabilities for multiple queries by compiling a temporary program.
248    pub(crate) fn evaluate_query_probabilities(&self, queries: &[String]) -> PyResult<Vec<f64>> {
249        #[cfg(not(feature = "host-io"))]
250        {
251            let _ = queries;
252            return Err(types::host_io_disabled_pyerr());
253        }
254
255        #[cfg(feature = "host-io")]
256        {
257            // Build source with queries appended
258            let mut source_with_queries = self._source.clone();
259            for query in queries {
260                source_with_queries.push_str(&format!("\nquery({}).", query));
261            }
262
263            // Compile and evaluate the temporary program
264            let result: Vec<QueryProbability> = match self._prob_engine {
265                xlog_logic::ast::ProbEngine::ExactDdnnf => {
266                    let program = ExactDdnnfProgram::compile_source_with_gpu(
267                        &source_with_queries,
268                        self._gpu_config,
269                    )
270                    .map_err(|e| types::gpu_err("Query compilation error", e))?;
271
272                    program
273                        .evaluate()
274                        .map_err(|e| types::gpu_err("Query evaluation error", e))?
275                        .query_probs
276                }
277                xlog_logic::ast::ProbEngine::Mc => {
278                    let program =
279                        McProgram::compile_source_with_gpu(&source_with_queries, self._gpu_config)
280                            .map_err(|e| types::gpu_err("Query compilation error", e))?;
281
282                    let cfg = resolve_mc_eval_config(
283                        &self.ast.directives,
284                        None,
285                        None,
286                        None,
287                        None,
288                        None,
289                        false,
290                    )
291                    .map_err(types::xlog_err)?;
292                    program
293                        .evaluate(cfg)
294                        .map_err(|e| types::gpu_err("Query evaluation error", e))?
295                        .query_estimates
296                        .into_iter()
297                        .map(|e| QueryProbability {
298                            atom: e.atom,
299                            prob: e.prob,
300                            log_prob: e.log_prob,
301                        })
302                        .collect()
303                }
304            };
305
306            // Extract probabilities in query order
307            // The results should be in the same order as queries were added
308            let probs: Vec<f64> = result.iter().map(|qp| qp.prob).collect();
309
310            if probs.len() != queries.len() {
311                return Err(PyRuntimeError::new_err(format!(
312                    "Expected {} query results, got {}",
313                    queries.len(),
314                    probs.len()
315                )));
316            }
317
318            Ok(probs)
319        }
320    }

323    fn pack_result_probs(
324        &self,
325        py: Python<'_>,
326        query_probs: Vec<QueryProbability>,
327        log_z_e: f64,
328    ) -> PyResult<EvalResult> {
329        let mut atoms: Vec<String> = Vec::with_capacity(query_probs.len());
330        let mut probs: Vec<f64> = Vec::with_capacity(query_probs.len());
331        let mut log_probs: Vec<f64> = Vec::with_capacity(query_probs.len());
332
333        for q in query_probs {
334            atoms.push(atom_to_string(&q.atom));

336            log_probs.push(q.log_prob);
337        }
338
339        let schema = Schema::new(vec![("col0".to_string(), ScalarType::F64)]);
340        let prob_buf = self
341            .output_provider
342            .create_buffer_from_slice::<f64>(&probs, schema.clone())
343            .map_err(types::xlog_err)?;
344        let log_prob_buf = self
345            .output_provider
346            .create_buffer_from_slice::<f64>(&log_probs, schema)
347            .map_err(types::xlog_err)?;
348
349        let prob_tensor = self
350            .output_provider
351            .to_dlpack_table(prob_buf)
352            .column(0)
353            .map_err(types::xlog_err)?;
354        let log_prob_tensor = self
355            .output_provider
356            .to_dlpack_table(log_prob_buf)
357            .column(0)
358            .map_err(types::xlog_err)?;
359
360        Ok(EvalResult {
361            atoms,
362            prob: dlpack_capsule_from_tensor(py, prob_tensor)?,
363            log_prob: dlpack_capsule_from_tensor(py, log_prob_tensor)?,
364            num_vars: self.program.num_vars(),

379            nonmonotone_iteration_limit_hits: None,

579#[pymethods]
580impl CompiledProgram {
581    #[pyo3(signature = (return_grads=false, samples=None, seed=None, confidence=None, max_nonmonotone_iterations=None, sampling_method=None, memory_mb=None, allow_cpu_oracle=false))]
582    #[expect(
583        clippy::too_many_arguments,
584        reason = "the Python evaluation API exposes each documented Monte Carlo override by name"
585    )]
586    pub fn evaluate(
587        &self,
588        py: Python<'_>,
589        return_grads: bool,
590        samples: Option<usize>,
591        seed: Option<u64>,
592        confidence: Option<f64>,
593        max_nonmonotone_iterations: Option<usize>,
594        sampling_method: Option<String>,
595        memory_mb: Option<u64>,
596        allow_cpu_oracle: bool,
597    ) -> PyResult<EvalResult> {
598        enforce_call_memory_limit(&self.output_provider, memory_mb)?;
599        match &self.program {
600            CompiledProbProgram::Exact(_program) => {
601                if samples.is_some()
602                    || seed.is_some()
603                    || confidence.is_some()
604                    || max_nonmonotone_iterations.is_some()
605                    || sampling_method.is_some()
606                    || allow_cpu_oracle
607                {
608                    return Err(PyValueError::new_err(
609                        "samples, seed, confidence, max_nonmonotone_iterations, \
610                         sampling_method, and allow_cpu_oracle are only supported for \
611                         prob_engine='mc'",
612                    ));
613                }
614                #[cfg(feature = "host-io")]
615                {
616                    if return_grads {
617                        let result = py
618                            .detach(|| _program.evaluate_gpu_with_grads())
619                            .map_err(types::xlog_err)?;
620                        self.pack_result_with_grads(py, result)
621                    } else {
622                        let result = py.detach(|| _program.evaluate()).map_err(types::xlog_err)?;
623                        self.pack_result_probs(py, result.query_probs, result.log_z_e)
624                    }
625                }
626                #[cfg(not(feature = "host-io"))]
627                {
628                    let _ = return_grads;
629                    Err(types::host_io_disabled_pyerr())
630                }
631            }
632            CompiledProbProgram::Mc(_program) => {
633                if return_grads {
634                    return Err(PyValueError::new_err(
635                        "MC inference does not support gradients (return_grads must be false)",
636                    ));
637                }
638
639                let cfg = resolve_mc_eval_config(
640                    &self.ast.directives,
641                    samples,
642                    seed,
643                    confidence,
644                    max_nonmonotone_iterations,
645                    Self::parse_sampling_method(sampling_method)?,
646                    allow_cpu_oracle,
647                )
648                .map_err(types::xlog_err)?;
649                #[cfg(feature = "host-io")]
650                {
651                    let result = py
652                        .detach(|| _program.evaluate(cfg))
653                        .map_err(types::xlog_err)?;
654                    self.pack_result_mc(py, result)
655                }
656                #[cfg(not(feature = "host-io"))]
657                {
658                    let _ = cfg;
659                    Err(types::host_io_disabled_pyerr())
660                }
661            }
662        }
663    }

665    /// Which probabilistic fact each CNF variable stands for.
666    ///
667    /// The returned list's length is the CNF encoder's variable *capacity*,
668    /// not the number of CNF variables in use and not the number of random
669    /// variables in the program — a real fraction of entries are
670    /// `{"kind": "other"}` padding (do not use `len()` of the result as a
671    /// variable count). Entry `i` describes CNF variable `i` — the same
672    /// position `i` that the `grad_true` / `grad_false` vectors of
673    /// `evaluate(return_grads=True)` use (index `0` is unused padding, since
674    /// CNF variables are 1-indexed).
675    ///
676    /// For a `"choice"` entry (one Bernoulli decision of an annotated
677    /// disjunction's chain), `probs[choice_index]` is the disjunction's
678    /// *declared, marginal* probability and is display context only; `prob`
679    /// is the *conditional* Bernoulli parameter actually assigned to this
680    /// variable's weight, and `prob * (1 - prob)` — not
681    /// `probs[choice_index] * (1 - probs[choice_index])` — is the correct
682    /// Jacobian for `grad_true` / `grad_false` at this position.
683    ///
684    /// Raises `ValueError` for Monte Carlo programs, and for exact programs
685    /// compiled through the GPU count-lift fast path (count aggregates
686    /// without evidence or annotated disjunctions), since that path never
687    /// builds a CNF encoding and therefore has no variable map to report —
688    /// this is *not* the same as the program having no probabilistic facts.
689    pub fn prob_var_map(&self, py: Python<'_>) -> PyResult<Vec<Py<PyAny>>> {
690        #[cfg(feature = "host-io")]
691        {
692            use xlog_prob::exact::ProbVarInfo;
693
694            let entries = match &self.program {
695                CompiledProbProgram::Exact(p) => {
696                    if p.uses_gpu_native_count_lift() {
697                        return Err(PyValueError::new_err(
698                            "prob_var_map is unavailable: this program was compiled through \
699                             the GPU count-lift fast path (a count aggregate without evidence \
700                             or annotated disjunctions), which evaluates the aggregate with a \
701                             dedicated kernel and never builds a CNF encoding. There is no \
702                             CNF-variable-to-fact map to report. This does not mean the \
703                             program has no probabilistic facts.",
704                        ));
705                    }
706                    p.prob_var_map()
707                }
708                CompiledProbProgram::Mc(_) => {
709                    return Err(PyValueError::new_err(
710                        "prob_var_map is only available for the exact engine",
711                    ))
712                }
713            };
714
715            let mut out: Vec<Py<PyAny>> = Vec::with_capacity(entries.len());
716            for entry in entries {
717                let d = PyDict::new(py);
718                match entry {
719                    ProbVarInfo::Fact { atom, prob } => {
720                        d.set_item("kind", "fact")?;
721                        d.set_item("atom", atom_to_string(&atom))?;
722                        d.set_item("prob", prob)?;
723                    }
724                    ProbVarInfo::Choice {
725                        choices,
726                        choice_index,
727                        prob,
728                    } => {
729                        d.set_item("kind", "choice")?;
730                        d.set_item(
731                            "atoms",
732                            choices
733                                .iter()
734                                .map(|(a, _)| atom_to_string(a))
735                                .collect::<Vec<_>>(),
736                        )?;
737                        // Declared, marginal probabilities of the whole disjunction
738                        // (display context only — probs[choice_index] is NOT this
739                        // variable's own Bernoulli parameter; see "prob" below).
740                        d.set_item(
741                            "probs",
742                            choices.iter().map(|(_, p)| *p).collect::<Vec<f64>>(),
743                        )?;
744                        d.set_item("choice_index", choice_index)?;
745                        // This chain variable's own (conditional) Bernoulli parameter,
746                        // i.e. the weight actually used by the GPU circuit for this CNF
747                        // variable. Use prob * (1 - prob), not
748                        // probs[choice_index] * (1 - probs[choice_index]), as the
749                        // Jacobian for grad_true/grad_false at this position.
750                        d.set_item("prob", prob)?;
751                    }
752                    ProbVarInfo::Other => {
753                        d.set_item("kind", "other")?;
754                    }
755                }
756                out.push(d.into());
757            }
758            Ok(out)
759        }
760        #[cfg(not(feature = "host-io"))]
761        {
762            let _ = py;
763            Err(types::host_io_disabled_pyerr())
764        }
765    }

767    /// Evaluate Monte Carlo programs and return device-only result counts via DLPack.
768    ///
769    /// This is the primary GPU-native API surface for MC inference. It never performs
770    /// device->host reads for result data (only returns device buffers).
771    #[pyo3(signature = (samples=None, seed=None, confidence=None, max_nonmonotone_iterations=None, sampling_method=None, memory_mb=None))]
772    #[expect(
773        clippy::too_many_arguments,
774        reason = "the GPU-native Python evaluation API exposes each documented override by name"
775    )]
776    pub fn evaluate_device(
777        &self,
778        py: Python<'_>,
779        samples: Option<usize>,
780        seed: Option<u64>,
781        confidence: Option<f64>,
782        max_nonmonotone_iterations: Option<usize>,
783        sampling_method: Option<String>,
784        memory_mb: Option<u64>,
785    ) -> PyResult<McDeviceEvalResult> {
786        enforce_call_memory_limit(&self.output_provider, memory_mb)?;
787        let (
788            query_counts,
789            evidence_count,
790            total_samples,
791            seed,
792            confidence,
793            nonmonotone_sccs,
794            nonmonotone_cycles,
795            nonmonotone_iteration_limit_hits,
796            sampling_method_val,
797            no_host,
798        ) = match &self.program {
799            CompiledProbProgram::Mc(program) => {
800                let cfg = resolve_mc_eval_config(
801                    &self.ast.directives,
802                    samples,
803                    seed,
804                    confidence,
805                    max_nonmonotone_iterations,
806                    Self::parse_sampling_method(sampling_method)?,
807                    false,
808                )
809                .map_err(types::xlog_err)?;
810
811                let provider = self.output_provider.clone();
812                let result = py
813                    .detach(|| program.evaluate_gpu_device_with_provider(cfg, provider))
814                    .map_err(types::xlog_err)?;
815
816                (
817                    result.query_counts,
818                    result.evidence_count,
819                    result.total_samples,
820                    result.seed,
821                    result.confidence,
822                    result.nonmonotone_sccs,
823                    result.nonmonotone_cycles,
824                    result.nonmonotone_iteration_limit_hits,
825                    result.sampling_method,
826                    result.no_host,
827                )
828            }
829            _ => {
830                return Err(PyValueError::new_err(
831                    "evaluate_device is only supported for prob_engine='mc'",
832                ))
833            }
834        };
835
836        // PyTorch does not support unsigned 32-bit types. Export as i32 (bitwise identical for
837        // counts < 2^31) for maximum DLPack consumer compatibility.
838        let schema_i32 = Schema::new(vec![("col0".to_string(), ScalarType::I32)]);
839
840        let make_count_tensor =
841            |counts: xlog_cuda::memory::TrackedCudaSlice<u32>, rows: u64| -> PyResult<Py<PyAny>> {
842                let rows_u32 = u32::try_from(rows).map_err(|_| {
843                    PyValueError::new_err(format!("Row count {} exceeds u32::MAX", rows))
844                })?;
845
846                let mut d_num_rows = self
847                    .output_provider
848                    .memory()
849                    .alloc::<u32>(1)
850                    .map_err(types::xlog_err)?;
851                self.output_provider
852                    .device()
853                    .inner()
854                    .htod_sync_copy_into(&[rows_u32], &mut d_num_rows)
855                    .map_err(types::xlog_err)?;
856
857                let buffer = xlog_cuda::CudaBuffer::from_columns(
858                    vec![counts.into_bytes().into()],
859                    rows,
860                    d_num_rows,
861                    schema_i32.clone(),
862                );
863                let tensor = self
864                    .output_provider
865                    .to_dlpack_table(buffer)
866                    .column(0)
867                    .map_err(types::xlog_err)?;
868                dlpack_capsule_from_tensor(py, tensor)
869            };
870
871        let query_rows = u64::try_from(query_counts.len())
872            .map_err(|_| PyValueError::new_err("query_counts length overflow"))?;
873        let query_counts_capsule = make_count_tensor(query_counts, query_rows)?;
874        let evidence_count_capsule = make_count_tensor(evidence_count, 1)?;
875        let resident_no_host_certified = no_host.is_no_host();
876
877        Ok(McDeviceEvalResult {
878            query_counts: query_counts_capsule,
879            evidence_count: evidence_count_capsule,
880            total_samples,
881            seed,
882            confidence,
883            nonmonotone_semantics: xlog_prob::mc::NONMONOTONE_SEMANTICS.to_string(),
884            nonmonotone_sccs,
885            nonmonotone_cycles,
886            nonmonotone_iteration_limit_hits,
887            sampling_method: match sampling_method_val {
888                McSamplingMethod::Rejection => "rejection".to_string(),
889                McSamplingMethod::EvidenceClamping => "evidence_clamping".to_string(),
890            },
891            resident_no_host_certified,
892            resident_no_host_policy_result: if resident_no_host_certified {
893                "certified".to_string()
894            } else {
895                "failed".to_string()
896            },
897            resident_no_host_tracked_dtoh_calls: no_host.tracked_dtoh_calls,
898            resident_no_host_tracked_htod_calls: no_host.tracked_htod_calls,
899            resident_no_host_host_loop_iterations: no_host.host_loop_iterations,
900            resident_no_host_per_sample_host_launches: no_host.per_sample_host_launches,
901            resident_no_host_untracked_metadata_reads: no_host.untracked_metadata_reads,
902            resident_no_host_engine_launches: no_host.engine_launches,
903            resident_no_host_host_fixpoint_iterations: no_host.host_fixpoint_iterations,
904            resident_no_host_per_operator_host_allocations: no_host.per_operator_host_allocations,
905        })
906    }

912    /// Compute negative log-likelihood loss for a single query.
913    ///
914    /// NLL loss = -log(P(query))
915    ///
916    /// This is the fundamental training objective for neural-symbolic programs.
917    /// Lower loss means higher probability of the query being true.
918    ///
919    /// # Arguments
920    /// * `query` - Query atom as string, e.g., "digit(0, 5)" or "path(1, 3)"
921    ///
922    /// # Returns
923    /// The NLL loss value (always non-negative, 0 for certain facts)
924    fn nll_loss(&self, query: &str) -> PyResult<f64> {
925        let prob = self.evaluate_query_probability(query)?;
926        Ok(types::nll_loss_value(prob))
927    }
928
929    /// Compute sum of NLL losses for a batch of queries.
930    ///
931    /// Batch loss = Σ -log(P(query_i))
932    ///
933    /// More efficient than calling nll_loss repeatedly as all queries
934    /// are compiled and evaluated together.
935    ///
936    /// # Arguments
937    /// * `queries` - List of query atoms as strings
938    ///
939    /// # Returns
940    /// Sum of individual NLL losses (0.0 for empty batch)
941    fn nll_loss_batch(&self, queries: Vec<String>) -> PyResult<f64> {
942        if queries.is_empty() {
943            return Ok(0.0);
944        }
945
946        let probs = self.evaluate_query_probabilities(&queries)?;
947        Ok(probs.iter().map(|&p| types::nll_loss_value(p)).sum())
948    }
949
950    /// Compute mean NLL loss for a batch of queries.
951    ///
952    /// Mean loss = (1/n) Σ -log(P(query_i))
953    ///
954    /// Useful for comparing loss across batches of different sizes.
955    ///
956    /// # Arguments
957    /// * `queries` - List of query atoms as strings (must be non-empty)
958    ///
959    /// # Returns
960    /// Mean of individual NLL losses
961    ///
962    /// # Errors
963    /// Returns error if queries is empty
964    fn nll_loss_mean(&self, queries: Vec<String>) -> PyResult<f64> {
965        if queries.is_empty() {
966            return Err(PyValueError::new_err(
967                "Cannot compute mean NLL loss for empty query batch",
968            ));
969        }
970
971        let probs = self.evaluate_query_probabilities(&queries)?;
972        let sum: f64 = probs.iter().map(|&p| types::nll_loss_value(p)).sum();
973        Ok(sum / probs.len() as f64)
974    }
975
976    /// Compute NLL loss and return as PyTorch tensor.
977    ///
978    /// Returns a scalar tensor that can participate in autograd.
979    /// Use this when you need gradients to flow back through the loss.
980    ///
981    /// # Arguments
982    /// * `query` - Query atom as string
983    ///
984    /// # Returns
985    /// PyTorch scalar tensor containing the loss value
986    fn nll_loss_tensor(&self, py: Python<'_>, query: &str) -> PyResult<Py<PyAny>> {
987        let loss = self.nll_loss(query)?;
988        types::create_torch_tensor(py, loss)
989    }

991    /// Compute batch NLL loss and return as PyTorch tensor.
992    ///
993    /// # Arguments
994    /// * `queries` - List of query atoms as strings
995    ///
996    /// # Returns
997    /// PyTorch scalar tensor containing the sum of losses
998    fn nll_loss_batch_tensor(&self, py: Python<'_>, queries: Vec<String>) -> PyResult<Py<PyAny>> {
999        let loss = self.nll_loss_batch(queries)?;
1000        types::create_torch_tensor(py, loss)
1001    }

1007    /// Zero gradients for all registered networks.
1008    ///
1009    /// This should be called at the start of each training iteration
1010    /// to clear accumulated gradients from previous iterations.
1011    pub fn zero_grad(&self, py: Python<'_>) -> PyResult<()> {
1012        for name in self.network_registry.names() {
1013            if let Some(handle) = self.network_registry.get(name) {
1014                if let Some(optimizer) = handle.optimizer() {
1015                    optimizer.call_method0(py, "zero_grad")?;
1016                }
1017            }
1018        }
1019        Ok(())
1020    }

1022    /// Perform optimizer step for all registered networks.
1023    ///
1024    /// This applies the accumulated gradients to update network parameters.
1025    /// Should be called after forward_backward().
1026    pub fn optimizer_step(&self, py: Python<'_>) -> PyResult<()> {
1027        for name in self.network_registry.names() {
1028            if let Some(handle) = self.network_registry.get(name) {
1029                if let Some(optimizer) = handle.optimizer() {
1030                    optimizer.call_method0(py, "step")?;
1031                }
1032            }
1033        }
1034        Ok(())
1035    }

1037    /// Clip gradient norms for all registered networks.
1038    ///
1039    /// Uses `torch.nn.utils.clip_grad_norm_`.
1040    pub fn clip_grad_norms(&self, py: Python<'_>, max_norm: f64) -> PyResult<()> {
1041        let clip_fn = py.import("torch.nn.utils")?.getattr("clip_grad_norm_")?;
1042        for name in self.network_registry.names() {
1043            if let Some(handle) = self.network_registry.get(name) {
1044                if let Some(module) = handle.module() {
1045                    let params = module.call_method0(py, "parameters")?;
1046                    clip_fn.call1((params, max_norm))?;
1047                }
1048            }
1049        }
1050        Ok(())
1051    }

1053    /// Step the learning rate scheduler.
1054    ///
1055    /// PyTorch schedulers expect at least one optimizer step before the first
1056    /// scheduler step. Call this after `optimizer_step()` (or after a training
1057    /// path that performs an optimizer step internally).
1058    ///
1059    /// If `network_name` is provided, steps only that network's scheduler.
1060    /// If `None` (default), steps all registered schedulers.
1061    #[pyo3(signature = (network_name=None))]
1062    fn scheduler_step(&self, py: Python<'_>, network_name: Option<&str>) -> PyResult<()> {
1063        match network_name {
1064            Some(name) => {
1065                let handle = self.network_registry.get(name).ok_or_else(|| {
1066                    pyo3::exceptions::PyValueError::new_err(format!(
1067                        "No network registered with name '{name}'"
1068                    ))
1069                })?;
1070                if let Some(scheduler) = handle.scheduler() {
1071                    scheduler.call_method0(py, "step")?;
1072                }
1073            }
1074            None => {
1075                for name in self.network_registry.names() {
1076                    if let Some(handle) = self.network_registry.get(name) {
1077                        if let Some(scheduler) = handle.scheduler() {
1078                            scheduler.call_method0(py, "step")?;
1079                        }
1080                    }
1081                }
1082            }
1083        }
1084        Ok(())
1085    }

1087    /// Get the current learning rate for a registered network.
1088    ///
1089    /// Reads `optimizer.param_groups[0]['lr']`.
1090    ///
1091    /// # Arguments
1092    /// * `network_name` - Name used in register_network()
1093    fn get_lr(&self, py: Python<'_>, network_name: &str) -> PyResult<f64> {
1094        let handle = self.network_registry.get(network_name).ok_or_else(|| {
1095            pyo3::exceptions::PyValueError::new_err(format!(
1096                "No network registered with name '{network_name}'"
1097            ))
1098        })?;
1099        let optimizer = handle.optimizer().ok_or_else(|| {
1100            pyo3::exceptions::PyValueError::new_err(format!(
1101                "Network '{network_name}' has no optimizer"
1102            ))
1103        })?;
1104        let param_groups = optimizer.getattr(py, "param_groups")?;
1105        let group0 = param_groups.call_method1(py, "__getitem__", (0i32,))?;
1106        let lr = group0.call_method1(py, "__getitem__", ("lr",))?;
1107        lr.extract(py)
1108    }

1110    /// Set the learning rate for a registered network.
1111    ///
1112    /// Writes to all `optimizer.param_groups[i]['lr']`.
1113    ///
1114    /// # Arguments
1115    /// * `network_name` - Name used in register_network()
1116    /// * `lr` - New learning rate value
1117    fn set_lr(&self, py: Python<'_>, network_name: &str, lr: f64) -> PyResult<()> {
1118        let handle = self.network_registry.get(network_name).ok_or_else(|| {
1119            pyo3::exceptions::PyValueError::new_err(format!(
1120                "No network registered with name '{network_name}'"
1121            ))
1122        })?;
1123        let optimizer = handle.optimizer().ok_or_else(|| {
1124            pyo3::exceptions::PyValueError::new_err(format!(
1125                "Network '{network_name}' has no optimizer"
1126            ))
1127        })?;
1128        let param_groups = optimizer.getattr(py, "param_groups")?;
1129        let num_groups: usize = param_groups.call_method0(py, "__len__")?.extract(py)?;
1130        for i in 0..num_groups {
1131            let group = param_groups.call_method1(py, "__getitem__", (i as i32,))?;
1132            group.call_method(py, "__setitem__", ("lr", lr), None)?;
1133        }
1134        Ok(())
1135    }

1212    /// Return warmup profiling data as a Python dict (or None if profiling disabled).
1213    ///
1214    /// When XLOG_WARMUP_PROFILE=1, returns a dict with:
1215    ///   - "ptx": PTX load timing breakdown
1216    ///   - "circuit": circuit compilation timing breakdown
1217    /// Returns None if profiling is not enabled or no data is available.
1218    fn warmup_breakdown(&self, py: Python<'_>) -> PyResult<Option<Py<PyAny>>> {
1219        let ptx_profile = self.output_provider.ptx_load_profile();
1220        // The neural forward path records `self.last_compile_profile`. For a
1221        // purely probabilistic/deterministic program no neural forward runs,
1222        // so fall back to the compile profile captured when the program's own
1223        // circuit was built (the cold D4-compile + CDCL-verify). Without this
1224        // fallback `warmup_breakdown()` returned None for non-neural programs,
1225        // hiding the verification-overhead split from benchmark isolation runs.
1226        let circuit_profile = self
1227            .last_compile_profile
1228            .as_ref()
1229            .or_else(|| match &self.program {
1230                CompiledProbProgram::Exact(p) => p.last_compile_profile(),
1231                CompiledProbProgram::Mc(_) => None,
1232            });
1233
1234        // Return None if neither profile is available.
1235        if ptx_profile.is_none() && circuit_profile.is_none() {
1236            return Ok(None);
1237        }
1238
1239        let result = PyDict::new(py);
1240
1241        if let Some(ptx) = ptx_profile {
1242            let ptx_dict = PyDict::new(py);
1243            ptx_dict.set_item("total_sec", ptx.total_sec)?;
1244            ptx_dict.set_item("cubin_loaded", ptx.cubin_loaded)?;
1245            ptx_dict.set_item("ptx_fallback", ptx.ptx_fallback)?;
1246            let per_module = PyDict::new(py);
1247            for (name, sec) in &ptx.per_module_sec {
1248                per_module.set_item(name, *sec)?;
1249            }
1250            ptx_dict.set_item("per_module_sec", per_module)?;
1251            result.set_item("ptx", ptx_dict)?;
1252        }
1253
1254        if let Some(circuit) = circuit_profile {
1255            let circuit_dict = PyDict::new(py);
1256            circuit_dict.set_item("gpu_cache_hit", circuit.gpu_cache_hit)?;
1257            circuit_dict.set_item("disk_cache_hit", circuit.disk_cache_hit)?;
1258            circuit_dict.set_item("d4_compile_sec", circuit.d4_compile_sec)?;
1259            circuit_dict.set_item("verify_sec", circuit.verify_sec)?;
1260            circuit_dict.set_item("smooth_sec", circuit.smooth_sec)?;
1261            circuit_dict.set_item("cache_store_sec", circuit.cache_store_sec)?;
1262            circuit_dict.set_item("free_var_mask_sec", circuit.free_var_mask_sec)?;
1263            circuit_dict.set_item("cnf_hash_sec", circuit.cnf_hash_sec)?;
1264            result.set_item("circuit", circuit_dict)?;
1265        }
1266
1267        Ok(Some(result.into()))
1268    }
