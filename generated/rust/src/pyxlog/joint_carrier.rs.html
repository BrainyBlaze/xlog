1// PyO3 binding for the XLOG-owned joint constraint carrier.
2//
3// The carrier owns every solver buffer on device; Python consumers
4// receive zero-copy DLPack views of the exact producer allocations
5// (ownership stays with xlog) and drive the cold-path session
6// lifecycle: allocate -> register_schema -> bind_signatures ->
7// solve_label_feasibility. The fuel meter lives inside the session,
8// so exhaustion saturates across Python calls — a retry repeats the
9// identical typed refusal instead of slipping work past the budget.
10
11use std::sync::Mutex;

13use pyo3::create_exception;
14use pyo3::exceptions::{PyRuntimeError, PyValueError};
15use pyo3::prelude::*;

17use xlog_cuda::dlpack::{export_slice_managed_tensor, DLDataType, K_DLFLOAT, K_DLUINT};
18use xlog_cuda::{
19    CarrierBufferId, CarrierError, CudaDevice, FuelMeter, JointConstraintCarrier as NativeCarrier,

23create_exception!(
24    pyxlog,
25    CarrierRefused,
26    PyRuntimeError,
27    "Typed refusal from the joint constraint carrier; the message names \
28     the exact refused precondition and its literals."
29);
30
31create_exception!(
32    pyxlog,
33    SolverResourceExhausted,
34    PyRuntimeError,
35    "The device fuel budget is exhausted; the refusal carries the exact \
36     spent/limit literals and repeats identically on retry."
37);
38
39fn carrier_err(err: CarrierError) -> PyErr {
40    match err {
41        CarrierError::Solver(SolverError::ResourceExhausted {
42            fuel_spent,
43            fuel_limit,
44        }) => SolverResourceExhausted::new_err(format!(
45            "solver fuel exhausted: spent {fuel_spent} of {fuel_limit} \
46             node expansions"
47        )),
48        other => CarrierRefused::new_err(other.to_string()),
49    }
50}

52struct CarrierState {
53    carrier: NativeCarrier,
54    fuel: FuelMeter,
55    device_id: i32,
56}

58/// Session handle over the device-resident joint constraint carrier.
59#[pyclass(unsendable)]
60pub(crate) struct JointConstraintCarrier {
61    state: Mutex<CarrierState>,
62}
63
64#[pymethods]
65impl JointConstraintCarrier {
66    #[new]
67    #[pyo3(signature = (device, entities, domain_lanes, candidates, labels, fuel_limit, max_arity=2))]
68    fn new(
69        device: usize,
70        entities: usize,
71        domain_lanes: usize,
72        candidates: usize,
73        labels: usize,
74        fuel_limit: u64,
75        max_arity: usize,
76    ) -> PyResult<Self> {
77        let cuda = CudaDevice::new(device)
78            .map_err(|e| PyRuntimeError::new_err(format!("CUDA device {device}: {e}")))?;
79        let cuda_device = std::sync::Arc::new(cuda);
80        let carrier = if max_arity == 2 {
81            NativeCarrier::allocate(cuda_device, entities, domain_lanes, candidates, labels)
82        } else {
83            NativeCarrier::allocate_with_max_arity(
84                cuda_device,
85                entities,
86                domain_lanes,
87                candidates,
88                labels,
89                max_arity,
90            )
91        }
92        .map_err(carrier_err)?;
93        Ok(Self {
94            state: Mutex::new(CarrierState {
95                carrier,
96                fuel: FuelMeter::new(fuel_limit),
97                device_id: device as i32,
98            }),
99        })
100    }

102    fn register_schema(&self, catalog_sha: &str, solver_identity: &str) -> PyResult<()> {
103        let mut state = self.state.lock().expect("carrier state lock");
104        state
105            .carrier
106            .register_schema(catalog_sha, solver_identity)
107            .map_err(carrier_err)
108    }

112        state
113            .carrier
114            .bind_signatures(&head_masks, &tail_masks)
115            .map_err(carrier_err)
116    }

126    /// Export one carrier buffer as a zero-copy DLPack capsule
127    /// (`torch.utils.dlpack.from_dlpack` consumes it directly). The
128    /// view shares the exact solve allocation; xlog stays the owner.
129    ///
130    /// Stream-ordering obligation: writes made through the exported
131    /// view happen on the consumer's stream. Either call
132    /// `note_producer_stream` with that stream's handle after the
133    /// writes (the solve then waits on device — the measured-region
134    /// path), or synchronize on the host before solving.
135    fn export_buffer(&self, py: Python<'_>, name: &str) -> PyResult<Py<PyAny>> {
136        let (id, dtype) = match name {
137            "domains" => (
138                CarrierBufferId::Domains,
139                DLDataType {
140                    code: K_DLUINT,
141                    bits: 64,
142                    lanes: 1,
143                },
144            ),
145            "scores" => (
146                CarrierBufferId::Scores,
147                DLDataType {
148                    code: K_DLFLOAT,
149                    bits: 32,
150                    lanes: 1,
151                },
152            ),
153            "constraints" => (
154                CarrierBufferId::Constraints,
155                DLDataType {
156                    code: K_DLUINT,
157                    bits: 32,
158                    lanes: 1,
159                },
160            ),
161            "argument_arities" => (
162                CarrierBufferId::ArgumentArities,
163                DLDataType {
164                    code: K_DLUINT,
165                    bits: 32,
166                    lanes: 1,
167                },
168            ),
169            "outputs" => (
170                CarrierBufferId::Outputs,
171                DLDataType {
172                    code: K_DLUINT,
173                    bits: 32,
174                    lanes: 1,
175                },
176            ),
177            "feasible_sets" => (
178                CarrierBufferId::FeasibleSets,
179                DLDataType {
180                    code: K_DLUINT,
181                    bits: 64,
182                    lanes: 1,
183                },
184            ),
185            "logical_counts" => (
186                CarrierBufferId::LogicalCounts,
187                DLDataType {
188                    code: K_DLUINT,
189                    bits: 32,
190                    lanes: 1,
191                },
192            ),
193            "map_results" => (
194                CarrierBufferId::MapResults,
195                DLDataType {
196                    code: K_DLUINT,
197                    bits: 32,
198                    lanes: 1,
199                },
200            ),
201            "solve_status" => (
202                CarrierBufferId::SolveStatus,
203                DLDataType {
204                    code: K_DLUINT,
205                    bits: 32,
206                    lanes: 1,
207                },
208            ),
209            other => {
210                return Err(PyValueError::new_err(format!(
211                    "unknown carrier buffer {other:?}; valid names: domains, \
212                     scores, constraints, argument_arities, outputs, feasible_sets, \
213                     logical_counts, map_results, solve_status"
214                )))
215            }
216        };
217        let state = self.state.lock().expect("carrier state lock");
218        let export = state.carrier.export_buffer(id);
219        let tensor = export_slice_managed_tensor(
220            export.slice,
221            state.device_id,
222            dtype,
223            export.rows,
224            export.cols,
225        )
226        .map_err(|e| PyRuntimeError::new_err(e.to_string()))?;
227        crate::dlpack_capsule_from_tensor(py, tensor)
228    }

230    /// Launch the on-device feasibility solve. Orders against
231    /// runtime-recorded events plus every producer event noted via
232    /// `note_producer_stream`; producer writes not covered by a noted
233    /// event or a host sync race the launch.
234    fn solve_label_feasibility(&self, abstain_label: u32) -> PyResult<()> {
235        let mut state = self.state.lock().expect("carrier state lock");
236        let state = &mut *state;
237        state
238            .carrier
239            .solve_label_feasibility(abstain_label, &mut state.fuel)
240            .map_err(carrier_err)
241    }
242
243    /// Launch the per-candidate exact top-two stage, consuming the
244    /// feasibility stage's feasible sets on device. Refuses typed
245    /// before feasibility has run. The same caller synchronization
246    /// obligation applies between stages: device writes through
247    /// exported views (domains, scores) after the feasibility solve
248    /// must be synchronized before this call. Results in
249    /// `map_results` are the global max-marginal ONLY for
250    /// single-candidate components; a set ambiguity flag must never
251    /// emit as a unique MAP label.
252    fn solve_label_map_top2(&self) -> PyResult<()> {
253        let mut state = self.state.lock().expect("carrier state lock");
254        let state = &mut *state;
255        state
256            .carrier
257            .solve_label_map_top2(&mut state.fuel)
258            .map_err(carrier_err)
259    }
260
261    /// Discover components from the carrier-owned argument matrix and
262    /// solve every multi-candidate component exactly on the device.
263    /// Per-row authority afterwards is `solve_status`:
264    /// 2 = exact enumeration, 4 = exact device-discovered chain DP,
265    /// 5 = exact general branch-and-bound, 3 = refused on measured
266    /// fuel exhaustion, 0xFFFFFFFF = poisoned;
267    /// untouched rows keep top-two authority.
268    fn solve_components_exact(&self) -> PyResult<()> {
269        let mut state = self.state.lock().expect("carrier state lock");
270        let state = &mut *state;
271        state
272            .carrier
273            .solve_components_exact(&mut state.fuel)
274            .map_err(carrier_err)
275    }
276
277    /// Record a producer-completion event on an external CUDA stream.
278    /// The next solve stage waits on every noted event on device
279    /// before launching, replacing any host synchronization between
280    /// producer writes through exported views and the solve. Use a
281    /// dedicated stream (`torch.cuda.Stream()`, pass its
282    /// `cuda_stream`): torch's default stream carries raw handle 0,
283    /// which is indistinguishable from a null handle and is refused.
284    fn note_producer_stream(&self, external_stream: u64) -> PyResult<()> {
285        let mut state = self.state.lock().expect("carrier state lock");
286        state
287            .carrier
288            .note_producer_stream(external_stream)
289            .map_err(carrier_err)
290    }
291
292    /// Register an external CUDA consumer stream for the next
293    /// successful solve stage. The carrier records completion on its
294    /// internal stream and enqueues a wait on every registered
295    /// consumer stream without a host synchronization barrier.
296    ///
297    /// Registrations are one-shot and are also cleared when the solve
298    /// fails or the carrier is dropped. Use a dedicated stream
299    /// (`torch.cuda.Stream()`, pass its `cuda_stream`); raw handle zero
300    /// is refused.
301    fn note_consumer_stream(&self, external_stream: u64) -> PyResult<()> {
302        let mut state = self.state.lock().expect("carrier state lock");
303        state
304            .carrier
305            .note_consumer_stream(external_stream)
306            .map_err(carrier_err)
307    }
308
309    #[getter]
310    fn fuel_spent(&self) -> u64 {
311        self.state.lock().expect("carrier state lock").fuel.spent()
312    }
